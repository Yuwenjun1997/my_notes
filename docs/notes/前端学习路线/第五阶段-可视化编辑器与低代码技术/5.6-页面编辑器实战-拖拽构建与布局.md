# 页面编辑器实战：拖拽构建与布局

## 从组件面板拖入画布

将 [dnd-kit 拖拽能力](./5.2-React拖拽技术-dnd-kit.md) 与 [编辑器架构](./5.5-可视化页面编辑器架构.md) 结合，实现从组件面板拖入画布生成新节点：

```tsx
import { DndContext, DragOverlay, closestCenter } from '@dnd-kit/core';
import { useDroppable } from '@dnd-kit/core';
import { nanoid } from 'nanoid';

function Editor() {
  const { schema, addNode, selectNode } = useEditorStore();
  const [activeMeta, setActiveMeta] = useState<ComponentMeta | null>(null);

  function handleDragStart({ active }) {
    // 记录正在拖拽的组件元数据，用于 DragOverlay 预览
    const meta = findComponentMeta(active.id);
    setActiveMeta(meta);
  }

  function handleDragEnd({ active, over }) {
    setActiveMeta(null);
    if (!over || over.id !== 'canvas') return;

    const meta = findComponentMeta(active.id);
    if (!meta) return;  // 画布内排序，非新组件拖入

    // 在画布末尾添加新节点
    const newNode: SchemaNode = {
      id: nanoid(),
      componentName: meta.componentName,
      props: { ...meta.defaultSchema.props },
      children: meta.defaultSchema.children,
    };
    addNode(newNode);
    selectNode(newNode.id);
  }

  return (
    <DndContext
      sensors={sensors}
      collisionDetection={closestCenter}
      onDragStart={handleDragStart}
      onDragEnd={handleDragEnd}
    >
      <ComponentPalette />
      <CanvasArea />
      <DragOverlay>
        {activeMeta ? <div className="drag-preview">{activeMeta.title}</div> : null}
      </DragOverlay>
    </DndContext>
  );
}

// 画布作为放置目标
function CanvasArea() {
  const { setNodeRef, isOver } = useDroppable({ id: 'canvas' });

  return (
    <div
      ref={setNodeRef}
      className={`canvas-area ${isOver ? 'drop-active' : ''}`}
      style={{ minHeight: 400, background: isOver ? '#f0f5ff' : '#fafafa' }}
    >
      <Canvas />
    </div>
  );
}
```

## 画布内排序与嵌套

让画布内的节点支持拖拽重排，并允许将组件拖入容器组件（如 Row、Card、Tabs）中：

```tsx
// 递归渲染画布节点，每个节点都是可排序的
function CanvasNode({ node, index }: { node: SchemaNode; index: number }) {
  const { attributes, listeners, setNodeRef, transform, transition } = useSortable({
    id: node.id,
  });

  const Component = registry[node.componentName];
  const isContainer = CONTAINER_COMPONENTS.includes(node.componentName);

  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
    position: 'relative' as const,
    minHeight: 40,
    outline: '1px dashed transparent',
    margin: 4,
  };

  return (
    <div ref={setNodeRef} style={style} {...attributes} {...listeners}>
      <Component {...node.props}>
        {isContainer && node.children?.map((child, i) => (
          <CanvasNode key={child.id} node={child} index={i} />
        ))}
      </Component>
      {/* 拖拽手柄 */}
      <div className="drag-handle">⋮⋮</div>
    </div>
  );
}
```

容器组件（如 Flex、Grid、Card）需要在注册时声明支持子节点：

```ts
const CONTAINER_COMPONENTS = ['Flex', 'Grid', 'Card', 'Tabs', 'Collapse'];

// 容器组件的 Schema 需要 children 字段
const flexDefaultSchema: SchemaNode = {
  componentName: 'Flex',
  props: { gap: 8, direction: 'row' },
  children: [],  // 空子节点列表，等待拖入填充
};
```

## 选择与高亮

点击节点时高亮选中，同时在右侧属性面板显示该节点的配置表单：

```tsx
// 全局状态
interface EditorState {
  selectedNodeId: string | null;
  hoveredNodeId: string | null;
  // ...其他状态
}

// 选中层覆盖在画布上方，显示蓝色边框
function SelectionOverlay({ schema }: { schema: SchemaNode }) {
  const { selectedNodeId } = useEditorStore();
  if (!selectedNodeId) return null;

  const nodeRect = getNodeRect(selectedNodeId);  // 通过 ref 获取节点位置
  if (!nodeRect) return null;

  return (
    <div
      className="selection-overlay"
      style={{
        position: 'absolute',
        top: nodeRect.top,
        left: nodeRect.left,
        width: nodeRect.width,
        height: nodeRect.height,
        border: '2px solid #1890ff',
        pointerEvents: 'none',
      }}
    >
      <div className="selection-label">
        {getNodeName(selectedNodeId)}
      </div>
    </div>
  );
}
```

## 属性面板实时编辑

属性面板变更实时反映到画布上，使用 [Zustand + Immer](./5.5-可视化页面编辑器架构.md) 进行不可变更新：

```tsx
function PropertyPanel() {
  const { selectedNodeId, schema, updateNodeProps } = useEditorStore();

  const selectedNode = selectedNodeId ? findNode(schema, selectedNodeId) : null;
  const meta = selectedNode ? getComponentMeta(selectedNode.componentName) : null;

  if (!selectedNode || !meta) {
    return <div className="empty-hint">请选择画布中的组件进行编辑</div>;
  }

  return (
    <div className="property-panel">
      <h4>{meta.title}</h4>
      {meta.propertySchema.map(field => (
        <div key={field.key} className="prop-field">
          <label>{field.label}</label>
          <PropEditor
            field={field}
            value={selectedNode.props?.[field.key]}
            onChange={(val) => updateNodeProps(selectedNode.id, { [field.key]: val })}
          />
        </div>
      ))}
    </div>
  );
}
```

## SchemaTree 树形面板

类似于浏览器 DevTools 的 Elements 面板，展示 Schema 的层级结构：

```tsx
function SchemaTree() {
  const { schema, selectedNodeId, selectNode } = useEditorStore();

  return (
    <div className="schema-tree">
      <h4>页面结构</h4>
      <TreeNode node={schema} depth={0} />
    </div>
  );
}

function TreeNode({ node, depth }: { node: SchemaNode; depth: number }) {
  const { selectedNodeId, selectNode } = useEditorStore();
  const [expanded, setExpanded] = useState(true);
  const hasChildren = node.children && node.children.length > 0;

  return (
    <div>
      <div
        className={`tree-node ${selectedNodeId === node.id ? 'selected' : ''}`}
        style={{ paddingLeft: depth * 16 }}
        onClick={() => selectNode(node.id)}
      >
        {hasChildren && (
          <span className="expand-toggle" onClick={e => { e.stopPropagation(); setExpanded(!expanded); }}>
            {expanded ? '▼' : '▶'}
          </span>
        )}
        <span className="node-icon">{getNodeIcon(node.componentName)}</span>
        <span className="node-name">
          {node.componentName}
          {node.props?.label ? ` · ${node.props.label}` : ''}
        </span>
      </div>
      {expanded && hasChildren && node.children.map(child => (
        <TreeNode key={child.id} node={child} depth={depth + 1} />
      ))}
    </div>
  );
}
```

树形面板与画布双向联动：点击树节点 → 画布中对应节点高亮；点击画布节点 → 树中对应项高亮并滚动到可视区域。
