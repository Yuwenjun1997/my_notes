# 可观测性

## 一、日志

### 1.1 Logback 配置

Logback 是 Spring Boot 默认的日志框架，是 Log4j 的升级版本。

**logback-spring.xml 完整配置**

```xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration scan="true" scanPeriod="60 seconds">

    <!-- 属性定义 -->
    <springProperty scope="context" name="appName" source="spring.application.name"/>
    <property name="LOG_HOME" value="/var/log/${appName}"/>
    <property name="LOG_PATTERN" value="%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n"/>

    <!-- 彩色控制台输出 -->
    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder>
            <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} %clr([%thread]){magenta} %clr(%-5level) %clr(%logger{36}){cyan} - %msg%n</pattern>
            <charset>UTF-8</charset>
        </encoder>
    </appender>

    <!-- 滚动文件输出（按天和大小） -->
    <appender name="FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
        <file>${LOG_HOME}/${appName}.log</file>
        <rollingPolicy class="ch.qos.logback.core.rolling.TimeBasedRollingPolicy">
            <!-- 按天滚动，压缩旧日志 -->
            <fileNamePattern>${LOG_HOME}/${appName}.%d{yyyy-MM-dd}.%i.gz</fileNamePattern>
            <!-- 保留 30 天 -->
            <maxHistory>30</maxHistory>
            <timeBasedFileNamingAndTriggeringPolicy class="ch.qos.logback.core.rolling.SizeAndTimeBasedFNATP">
                <!-- 单个文件最大 100MB -->
                <maxFileSize>100MB</maxFileSize>
            </timeBasedFileNamingAndTriggeringPolicy>
        </rollingPolicy>
        <encoder>
            <pattern>${LOG_PATTERN}</pattern>
            <charset>UTF-8</charset>
        </encoder>
    </appender>

    <!-- 错误日志单独文件 -->
    <appender name="ERROR_FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
        <file>${LOG_HOME}/${appName}-error.log</file>
        <filter class="ch.qos.logback.classic.filter.LevelFilter">
            <level>ERROR</level>
            <onMatch>ACCEPT</onMatch>
            <onMismatch>DENY</onMismatch>
        </filter>
        <rollingPolicy class="ch.qos.logback.core.rolling.TimeBasedRollingPolicy">
            <fileNamePattern>${LOG_HOME}/${appName}-error.%d{yyyy-MM-dd}.log</fileNamePattern>
            <maxHistory>60</maxHistory>
        </rollingPolicy>
        <encoder>
            <pattern>${LOG_PATTERN}</pattern>
            <charset>UTF-8</charset>
        </encoder>
    </appender>

    <!-- 异步输出（提高性能） -->
    <appender name="ASYNC" class="ch.qos.logback.classic.AsyncAppender">
        <queueSize>512</queueSize>
        <discardingThreshold>0</discardingThreshold>
        <includeCallerData>false</includeCallerData>
        <appender-ref ref="FILE"/>
    </appender>

    <!-- 不同包日志级别配置 -->
    <logger name="com.example" level="DEBUG"/>
    <logger name="org.springframework" level="INFO"/>
    <logger name="org.springframework.web" level="INFO"/>
    <logger name="com.alibaba.nacos" level="WARN"/>
    <logger name="io.lettuce" level="WARN"/>

    <!-- 根日志级别 -->
    <root level="INFO">
        <appender-ref ref="CONSOLE"/>
        <appender-ref ref="ASYNC"/>
        <appender-ref ref="ERROR_FILE"/>
    </root>
</configuration>
```

**结构化日志（推荐生产环境使用）**

```java
@Component
@Slf4j
public class StructuredLoggingService {

    /**
     * 使用 MDC（Mapped Diagnostic Context）添加结构化字段
     */
    public void processOrder(Order order) {
        try {
            // 设置上下文信息（自动附加到当前线程的所有日志）
            MDC.put("orderId", order.getId().toString());
            MDC.put("userId", order.getUserId().toString());
            MDC.put("traceId", TraceContext.getTraceId());
            MDC.put("spanId", TraceContext.getSpanId());

            log.info("开始处理订单");
            // ... 业务处理 ...
            log.info("订单处理完成，金额: {}", order.getAmount());
            this.saveOrder(order);
        } catch (Exception e) {
            log.error("订单处理失败，错误信息: ", e);
        } finally {
            // 清理 MDC
            MDC.clear();
        }
    }
}
```

**日志最佳实践**

```java
// ✅ 正确：使用参数占位符
log.info("用户 {} 登录成功，IP: {}", userId, ipAddress);

// ❌ 错误：字符串拼接（即使日志级别不输出也会执行拼接）

// ✅ 正确：debug 日志加条件判断
if (log.isDebugEnabled()) {
    log.debug("查询参数: {}", JSON.toJSONString(params));
}

// ✅ 正确：异常日志传入异常对象
try {
    // ...
} catch (Exception e) {
    log.error("处理订单失败，orderId={}", orderId, e);  // e 作为参数，自动打印堆栈
}

// ❌ 错误：只传消息不传异常
log.error("处理订单失败: " + e.getMessage());  // 丢失堆栈信息
```

### 1.2 ELK 日志收集

ELK 是 Elasticsearch + Logstash + Kibana 的组合，用于日志的收集、存储和可视化。

**架构**

```text
微服务 → Filebeat（日志收集器）→ Logstash（日志处理）→ Elasticsearch（存储）→ Kibana（可视化）
  ↓
  应用输出日志到文件
  ↓
  Filebeat 监听日志文件变化
  ↓
  发送到 Logstash 或直接到 Elasticsearch
  ↓
  Kibana 查询和分析日志
```

**Filebeat 配置**

```yaml
# filebeat.yml
filebeat.inputs:
  - type: log
    enabled: true
    paths:
      - /var/log/*.log
    fields:
      app: user-service
      env: production
    multiline:
      pattern: '^\d{4}-\d{2}-\d{2}'  # 匹配时间戳开头
      negate: true
      match: after                    # 将非时间戳开头的行合并到上一行（含堆栈的异常日志）

output.logstash:
  hosts: ["logstash:5044"]
```

**Logstash 配置**

```ruby
# logstash.conf
input {
  beats {
    port => 5044
  }
}

filter {
  # 解析日志格式
  grok {
    match => { "message" => "%{TIMESTAMP_ISO8601:timestamp} \[%{DATA:thread}\] %{LOGLEVEL:level} %{DATA:logger} - %{GREEDYDATA:message}" }
  }

  # 添加时间戳
  date {
    match => ["timestamp", "yyyy-MM-dd HH:mm:ss.SSS"]
    target => "@timestamp"
  }

  # JSON 解析（如果日志内容是 JSON）
  json {
    source => "message"
    skip_on_invalid_json => true
  }

  # 添加字段
  mutate {
    add_field => { "environment" => "production" }
    remove_field => ["host", "beat"]
  }
}

output {
  elasticsearch {
    hosts => ["http://elasticsearch:9200"]
    index => "app-logs-%{+YYYY.MM.dd}"
  }
}
```

## 二、链路追踪

### 2.1 为什么需要链路追踪

在微服务架构中，一个请求会跨多个服务。当请求出错时，需要知道：

```text
用户请求 → API 网关 → 用户服务 → 订单服务 → 支付服务
                                              ↓
                                            数据库超时？

问题：到底是哪个服务慢？哪个环节出错了？
答案：链路追踪 —— 把整个请求链路的耗时和状态串联起来
```

**核心概念**

```text
Trace（追踪链）：
[Span A (API Gateway)]         ← 整个请求的根 Span
    ↓
[Span B (User Service)]        ← 子 Span
    ↓
[Span C (Order Service)]       ← 子 Span
    ↓
    ├─ [Span D (DB Query)]     ← 数据库调用
    └─ [Span E (Redis Get)]    ← Redis 调用
```

| 概念 | 说明 |
|------|------|
| Trace | 一次完整的请求链路，由 TraceId 标识 |
| Span | 链路中的一次操作，由 SpanId 标识 |
| ParentSpan | 父级 Span，表示调用关系 |

### 2.2 SkyWalking 配置与使用

Apache SkyWalking 是国产的开源 APM（应用性能监控）系统。

**架构**

```text
微服务（挂载 Agent）→ 通过 gRPC/HTTP 上报数据 → SkyWalking OAP Server → 存储（Elasticsearch/H2）
                                                        ↓
                                                   SkyWalking UI（可视化）
```

**Java Agent 方式接入（无侵入）**

```bash
# 1. 下载 SkyWalking Agent
# wget https://dlcdn.apache.org/skywalking/java-agent/9.0.0/apache-skywalking-java-agent-9.0.0.tgz

# 2. 启动时添加 Agent 参数
java -javaagent:/path/to/skywalking-agent/skywalking-agent.jar \
     -DSW_AGENT_NAME=user-service \          # 服务名
     -DSW_AGENT_COLLECTOR_BACKEND_SERVICES=127.0.0.1:11800 \  # OAP 地址
     -jar user-service.jar
```

**Docker 部署 SkyWalking**

**Spring Boot 集成（通过 SkyWalking API）**

```xml
<dependency>
    <groupId>org.apache.skywalking</groupId>
    <artifactId>apm-toolkit-trace</artifactId>
    <version>9.0.0</version>
```

```java
@Service
public class OrderService {

    // 使用 @Trace 标记需要追踪的方法
    @Trace(operationName = "OrderService.createOrder")
    public Order createOrder(OrderRequest request) {
        // 添加自定义标签
        ActiveSpan.tag("order_amount", request.getAmount().toString());
        ActiveSpan.tag("user_id", request.getUserId().toString());

        try {
            // ... 业务逻辑 ...
            return order;
        } catch (Exception e) {
            // 记录异常
            ActiveSpan.error(e);
            throw e;
        }
    }

    @Trace
    public void processPayment(Long orderId, BigDecimal amount) {
        // 创建子 Span
        AbstractSpan span = Tracer.createLocalSpan("payment.execute");
        try {
            span.setOperationName("调用支付网关");
            span.tag("order_id", orderId.toString());
            span.tag("amount", amount.toString());

            // 调用支付网关
            paymentService.pay(orderId, amount);

            span.asyncFinish();
        } catch (Exception e) {
            span.error(e);
            span.asyncFinish();
            throw e;
        }
    }
}
```

**自定义指标**

```java
@Component
public class CustomMetrics {

    private final Counter orderCreateCounter;
    private final Counter orderFailCounter;
    private final Timer orderProcessTimer;
    private final Gauge activeUserGauge;

    public CustomMetrics(MeterRegistry registry) {
        // 计数器：订单创建数
        orderCreateCounter = Counter.builder("orders.created.total")
                .description("订单创建总数")
                .tag("status", "success")
                .register(registry);

        // 计数器：订单失败数
        orderFailCounter = Counter.builder("orders.created.total")
                .description("订单创建失败数")
                .tag("status", "failed")
                .register(registry);

        // 计时器：订单处理耗时
        orderProcessTimer = Timer.builder("orders.process.duration")
                .description("订单处理耗时（毫秒）")
                .publishPercentiles(0.5, 0.95, 0.99)  // P50, P95, P99
                .register(registry);

        // 仪表：当前活跃用户数
        activeUserGauge = Gauge.builder("users.active.current", this,
                CustomMetrics::getActiveUserCount)
                .description("当前活跃用户数")
                .register(registry);
    }

    // 记录订单创建
    public void recordOrderCreated(boolean success) {
        if (success) {
            orderCreateCounter.increment();
        } else {
            orderFailCounter.increment();
        }
    }

    // 记录订单处理耗时
    public void recordOrderProcessTime(long millis) {
        orderProcessTimer.record(millis, TimeUnit.MILLISECONDS);
    }

    private int getActiveUserCount() {
        // 从 Redis 获取当前活跃用户数
        return redisService.countActiveUsers();
    }
}
```

### 3.2 Grafana 可视化

**创建 Dashboard 的 PromQL 查询**

| 可视化内容 | PromQL 查询 | 图表类型 |
|-----------|-------------|---------|
| 请求速率 | `rate(http_requests_total[1m])` | 时序图 |
| 响应时间 P99 | `http_server_requests_seconds{quantile="0.99"}` | 时序图 |
| 错误率 | `rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) * 100` | 时序图 |
| JVM 堆内存 | `jvm_memory_used_bytes{area="heap"}` | 面积图 |
| CPU 使用率 | `rate(process_cpu_usage[5m])` | 仪表盘 |
| 活跃用户 | `users_active_current` | 统计图 |

**关键监控指标**

```text
应用层面：
├── 请求速率（RPS）
├── 响应时间（P50/P95/P99）
├── 错误率（5xx 比例）
├── 活跃线程数
└── 数据库连接池使用率

JVM 层面：
├── 堆内存使用（Eden/Survivor/Old）
├── GC 频率和耗时（Young GC / Full GC）
├── 线程数量（活跃/阻塞/等待）
└── CPU 使用率

系统层面：
├── 磁盘 I/O（读写速率）
├── 网络 I/O
├── 内存使用
└── 文件描述符数量
```

**告警规则**

```yaml
# alert-rules.yml
groups:
  - name: spring-boot-alerts
    rules:
      # 错误率过高
      - alert: HighErrorRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) * 100 > 5
        for: 3m
        labels:
          severity: critical
        annotations:
          summary: "服务 {{ $labels.application }} 错误率超过 5%"
          description: "当前错误率: {{ $value }}%"

      # 响应时间过长
      - alert: SlowResponseTime
        expr: http_server_requests_seconds{quantile="0.99"} > 5
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "服务 {{ $labels.application }} P99 响应时间超过 5 秒"

      # Full GC 频繁
      - alert: FrequentFullGC
        expr: rate(jvm_gc_pause_seconds_count{action="end of major pause"}[5m]) > 0.1
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "服务 {{ $labels.application }} Full GC 频率过高"
          description: "每分钟 {{ $value }} 次 Full GC"

      # 实例宕机
      - alert: InstanceDown
        expr: up == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "服务 {{ $labels.application }} 实例 {{ $labels.instance }} 宕机"
```

## 四、实践项目

### 项目：完整可观测性平台搭建

**目标**：为微服务系统搭建完整的可观测性平台。

**要求**：
1. **日志**：Logback 输出结构化日志 → Filebeat → Logstash → Elasticsearch → Kibana
2. **链路追踪**：SkyWalking Agent 无侵入接入 → 可视化调用链
3. **指标监控**：Actuator → Prometheus → Grafana Dashboard
4. **告警**：Prometheus Alertmanager 配置告警规则

**架构部署**：

```text
                                                    ┌──────────┐
微服务A ──→ Logback ──→ 文件 ──→ Filebeat ──────→ │ Logstash │
微服务B ──→ Logback ──→ 文件 ──→ Filebeat ──────→ │          │
微服务C ──→ Logback ──→ 文件 ──→ Filebeat ──────→ │          │──→ Elasticsearch ──→ Kibana
                                                    └──────────┘
```
