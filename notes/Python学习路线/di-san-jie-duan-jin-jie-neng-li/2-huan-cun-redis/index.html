一、redis-py 基础

3.2.1 连接与基本操作

redis-py：Python 官方推荐的 Redis 客户端库。

# 安装
pip install redis

基本连接与命令：

import redis

# 创建客户端（会自动维护连接池）
r = redis.Redis(host="localhost", port=6379, db=0, password=None, decode_responses=True)

# 字符串操作
r.set("name", "张三")
r.setex("token", 3600, "abc123")   # ✅ 带过期时间的字符串
print(r.get("name"))               # 张三
print(r.ttl("token"))              # 剩余过期秒数

# 自增自减（适合计数）
r.incr("page_views")
r.incrby("page_views", 5)
print(r.get("page_views"))

# 哈希操作（对象存储）
r.hset("user:1001", mapping={"name": "李四", "age": 28})
print(r.hgetall("user:1001"))      # {'name': '李四', 'age': '28'}

# 列表（队列场景）
r.rpush("task_queue", "task-1", "task-2")
r.lpush("task_queue", "task-0")
print(r.lrange("task_queue", 0, -1))

# 集合 / 有序集合
r.sadd("tags:python", "web", "async", "cache")
r.zadd("ranking", {"alice": 100, "bob": 90})
print(r.zrevrange("ranking", 0, -1, withscores=True))

3.2.2 Pipeline 与事务

Pipeline（管道）：将多条命令打包一次发送，减少网络往返（RTT）。

import redis

r = redis.Redis(decode_responses=True)

# ✅ 批量操作，性能提升数倍
pipe = r.pipeline(transaction=True)   # transaction=True 启用事务
pipe.set("k1", "v1")
pipe.set("k2", "v2")
pipe.incr("counter")
results = pipe.execute()              # 一次性提交，返回结果列表

事务（Transaction）：保证多条命令原子执行。

方式特点冲突处理
MULTI/EXEC经典事务，命令排队后原子执行需 WATCH 处理冲突
Lua 脚本服务端原子执行，性能最好脚本内保证原子
pipeline(transaction=True)结合管道与事务简单场景够用

WATCH 乐观锁（CAS 模式）：

import redis

def transfer(r, from_id, to_id, amount):
    with r.pipeline() as pipe:
        while True:
            try:
                pipe.watch(f"balance:{from_id}")
                balance = int(pipe.get(f"balance:{from_id}"))
                if balance < amount:
                    raise ValueError("余额不足")
                pipe.multi()                       # 开始事务
                pipe.decrby(f"balance:{from_id}", amount)
                pipe.incrby(f"balance:{to_id}", amount)
                pipe.execute()                     # 若期间 key 被改，抛 WatchError
                break
            except redis.WatchError:
                continue  # 冲突则重试

二、缓存模式

3.2.3 functools.lru_cache

lru_cache：标准库内置的函数级缓存，适合纯函数（相同入参返回相同结果）。

from functools import lru_cache

@lru_cache(maxsize=128)   # 最多缓存 128 个结果，超限淘汰最久未用
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(100))  # ✅ 指数递归变为线性，毫秒级返回
print(fibonacci.cache_info())  # CacheInfo(hits=..., misses=..., ...)

fibonacci.cache_clear()  # 清空缓存

注意：lru_cache 只适用于无副作用、结果稳定的函数；参数必须是可哈希的。

3.2.4 接口缓存模式

缓存读写模式（Cache Aside）：

import redis
import json

r = redis.Redis(decode_responses=True)

def get_product(product_id: int) -> dict:
    """查询商品详情，优先读缓存"""
    cache_key = f"product:{product_id}"

    # 1. 先查缓存
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)

    # 2. 缓存未命中，查数据库
    product = query_db(product_id)   # 伪代码：数据库查询

    # 3. 回填缓存（带过期时间）
    r.setex(cache_key, 3600, json.dumps(product, ensure_ascii=False))
    return product

更新数据库后保持缓存一致：

def update_product(product_id, data):
    # 先更新数据库
    update_db(product_id, data)
    # 再删除缓存（下次查询重建）
    r.delete(f"product:{product_id}")

三、三大缓存问题

3.2.5 缓存穿透

缓存穿透：查询不存在的数据，缓存和数据库都没有，请求直接打到数据库。

解决方案：

方案原理缺点
空值缓存不存在的数据也缓存空值（短 TTL）占用内存，可能有短暂不一致
布隆过滤器请求前先过滤不存在的主键有误判率，实现较复杂
参数校验非法参数直接拒绝只能拦截部分请求
def get_user(user_id: int):
    cache_key = f"user:{user_id}"
    cached = r.get(cache_key)
    if cached is not None:
        return json.loads(cached)

    user = query_db(user_id)
    if user is None:
        # ✅ 缓存空值，设置较短过期时间，防止穿透
        r.setex(cache_key, 60, json.dumps(None))
        return None

    r.setex(cache_key, 3600, json.dumps(user))
    return user

3.2.6 缓存击穿

缓存击穿：热点 key 过期瞬间，大量请求同时打到数据库。

解决方案：互斥锁 + 单线程重建。

import redis
import json
import time
import uuid

r = redis.Redis(decode_responses=True)
LOCK_PREFIX = "lock:product:"
LOCK_TTL = 5  # 锁持有时间（秒）

def get_hot_product(product_id: int) -> dict:
    cache_key = f"product:{product_id}"
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)

    # 未命中：获取分布式锁，只让一个请求重建缓存
    lock_key = LOCK_PREFIX + str(product_id)
    lock_token = str(uuid.uuid4())
    acquired = r.set(lock_key, lock_token, nx=True, ex=LOCK_TTL)
    if acquired:
        try:
            # 双重检查：等待期间可能已被其他线程重建
            cached = r.get(cache_key)
            if cached:
                return json.loads(cached)
            product = query_db(product_id)
            r.setex(cache_key, 3600, json.dumps(product))
            return product
        finally:
            # ✅ 用 Lua 脚本保证"校验 token + 删除锁"原子性
            if r.get(lock_key) == lock_token:
                r.delete(lock_key)
    else:
        # 未抢到锁：短暂等待后重试
        time.sleep(0.05)
        return get_hot_product(product_id)

3.2.7 缓存雪崩

缓存雪崩：大量 key 在同一时间过期（或 Redis 宕机），导致数据库压力激增。

解决方案：

方案说明
随机过期时间过期时间加随机值，避免集中失效
多级缓存本地缓存（进程内）+ Redis 兜底
熔断限流数据库压力大时降级，返回旧数据
Redis 高可用主从 + 哨兵 + 集群，避免单点
import random

def cache_with_jitter(key, value, base_ttl=3600):
    """✅ 基础 TTL + 随机抖动，打散过期时间"""
    jitter = random.randint(0, 300)
    r.setex(key, base_ttl + jitter, value)

多级缓存架构：

from cachetools import TTLCache
import json

# 一级缓存：进程内本地缓存（最快，仅限单实例）
local_cache = TTLCache(maxsize=1000, ttl=60)

def get_with_multi_level(product_id: int):
    cache_key = f"product:{product_id}"

    # 第一级：本地内存
    if cache_key in local_cache:
        return local_cache[cache_key]

    # 第二级：Redis 分布式缓存
    cached = r.get(cache_key)
    if cached:
        data = json.loads(cached)
        local_cache[cache_key] = data  # 回填一级
        return data

    # 第三级：数据库（带锁防击穿）

四、分布式锁

3.2.8 分布式锁原理与实现

分布式锁：在多个应用实例之间互斥访问共享资源，是解决"缓存击穿"、"接口防重"、"秒杀扣库存"的基础设施。

核心要求：

  • 互斥性：任意时刻只有一个客户端持有锁
  • 防死锁：持锁客户端崩溃后锁能自动释放（过期时间）
  • 安全性：只能释放自己持有的锁（防止误删别人的锁）

基于 SETNX + Lua 的完整实现：

import redis
import uuid
import time

class RedisLock:
    def __init__(self, r: redis.Redis, name: str, timeout: int = 10):
        self.r = r
        self.name = f"lock:{name}"
        self.timeout = timeout
        self.token = str(uuid.uuid4())  # 唯一标识，用于安全释放

    # 释放锁的 Lua 脚本：校验 token 再删除，保证原子性
    UNLOCK_LUA = """
    if redis.call('get', KEYS[1]) == ARGV[1] then
        return redis.call('del', KEYS[1])
    else
        return 0
    end
    """

    def acquire(self, retries: int = 3) -> bool:
        for _ in range(retries):
            # nx=True：不存在才设置；ex：过期时间防死锁
            if self.r.set(self.name, self.token, nx=True, ex=self.timeout):
                return True
            time.sleep(0.05)  # 重试间隔
        return False

    def release(self) -> bool:
        # ✅ 原子操作：先比对 token，一致才删除
        return bool(self.r.eval(self.UNLOCK_LUA, 1, self.name, self.token))

    def __enter__(self):
        self.acquire()
        return self

    def __exit__(self, *args):
        self.release()

# 使用示例
lock = RedisLock(r, "seckill:sku:1001", timeout=5)
if lock.acquire():
    try:
        # 临界区：扣库存、下单等
        pass
    finally:
        lock.release()

Redlock 算法：针对 Redis 主从切换场景，向 N 个独立 Redis 节点申请锁，多数成功才算加锁成功。生产环境通常用 redlock-py 库或直接依赖 Redis Cluster 的高可用，而非纯 Redlock。

锁的自动续期：长时间任务可能超过锁过期时间，需要看门狗续期。

import threading

class RenewLock:
    """锁自动续期：每 1/3 过期时间续一次"""
    def __init__(self, r, name, timeout=10):
        self.r = r
        self.name = f"lock:{name}"
        self.timeout = timeout
        self.token = str(uuid.uuid4())
        self._stop = False

    def _renew_loop(self):
        def loop():
            while not self._stop:
                time.sleep(self.timeout / 3)
                lua = "if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('pexpire', KEYS[1], ARGV[2]) else return 0 end"
                self.r.eval(lua, 1, self.name, self.token, self.timeout * 1000)
        self._thread = threading.Thread(target=loop, daemon=True)
        self._thread.start()

    def acquire(self):
        if self.r.set(self.name, self.token, nx=True, ex=self.timeout):
            self._renew_loop()
            return True
        return False

    def stop(self):
        self._stop = True

五、异步 Redis 与 FastAPI 集成

3.2.9 redis.asyncio

异步客户端：在 asyncio 应用中使用，避免阻塞事件循环。

import asyncio
import redis.asyncio as aioredis

async def main():
    r = aioredis.Redis(host="localhost", port=6379, decode_responses=True)
    await r.set("k", "v")
    print(await r.get("k"))

    # ✅ 异步管道
    async with r.pipeline(transaction=True) as pipe:
        await pipe.set("k1", "v1")
        await pipe.set("k2", "v2")
        await pipe.execute()

    await r.aclose()  # 关闭连接

asyncio.run(main())

3.2.10 FastAPI 依赖注入 Redis

在 FastAPI 中管理 Redis 生命周期：

from fastapi import FastAPI, Depends
from contextlib import asynccontextmanager
import redis.asyncio as aioredis

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 应用启动时创建连接池
    app.state.redis = aioredis.from_url(
        "redis://localhost:6379/0",
        decode_responses=True,
        max_connections=50,
    )
    yield
    # 应用关闭时释放
    await app.state.redis.aclose()

app = FastAPI(lifespan=lifespan)

async def get_redis():
    return app.state.redis

@app.get("/count/{user_id}")
async def get_count(user_id: int, r=Depends(get_redis)):
    key = f"count:{user_id}"
    await r.incr(key)
    return {"count": int(await r.get(key))}

缓存装饰器（异步版）：

import json
import functools

def cache_result(key_prefix: str, ttl: int = 3600):
    def decorator(func):
        @functools.wraps(func)
        async def wrapper(*args, r=None, **kwargs):
            if r is None:
                return await func(*args, **kwargs)

            cache_key = f"{key_prefix}:{hash(args)}"
            cached = await r.get(cache_key)
            if cached:
                return json.loads(cached)

            result = await func(*args, **kwargs)
            await r.setex(cache_key, ttl, json.dumps(result))
            return result
        return wrapper
    return decorator
