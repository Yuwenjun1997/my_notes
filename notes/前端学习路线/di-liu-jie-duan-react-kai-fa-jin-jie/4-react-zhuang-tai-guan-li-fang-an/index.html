React状态管理方案

约 2895 字大约 10 分钟

2026-08-20

深入探讨 React 应用中的状态管理策略，从组件内状态到全局状态管理，对比 Context、Zustand、Jotai、Redux Toolkit 等方案的特点和适用场景。本文旨在帮助开发者根据项目需求选择最合适的状态管理方案。

状态管理层次

组件内状态

使用 useState 或 useReducer 管理组件内部状态。

// useState 适用于简单状态
function Counter() {
  const [count, setCount] = useState(0);
  
  return (
    <div>
      <p>计数：{count}</p>
      <button onClick={() => setCount(prev => prev + 1)}>增加</button>
      <button onClick={() => setCount(prev => prev - 1)}>减少</button>
      <button onClick={() => setCount(0)}>重置</button>
    </div>
  );
}

// useReducer 适用于复杂状态逻辑
function todoReducer(state, action) {
  switch (action.type) {
    case 'ADD_TODO':
      return [
        ...state,
        {
          id: Date.now(),
          text: action.payload,
          completed: false
        }
      ];
    
    case 'TOGGLE_TODO':
      return state.map(todo =>
        todo.id === action.payload
          ? { ...todo, completed: !todo.completed }
          : todo
      );
    
    case 'DELETE_TODO':
      return state.filter(todo => todo.id !== action.payload);
    
    case 'EDIT_TODO':
      return state.map(todo =>
        todo.id === action.payload.id
          ? { ...todo, text: action.payload.text }
          : todo
      );
    
    default:
      return state;
  }
}

function TodoApp() {
  const [todos, dispatch] = useReducer(todoReducer, []);
  const [inputValue, setInputValue] = useState('');
  
  const handleSubmit = (e) => {
    e.preventDefault();
    if (inputValue.trim()) {
      dispatch({ type: 'ADD_TODO', payload: inputValue.trim() });
      setInputValue('');
    }
  };
  
  return (
    <div>
      <form onSubmit={handleSubmit}>
        <input
          type="text"
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          placeholder="添加新任务"
        />
        <button type="submit">添加</button>
      </form>
      
      <ul>
        {todos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => dispatch({ type: 'TOGGLE_TODO', payload: todo.id })}
            />
            <span style={{ 
              textDecoration: todo.completed ? 'line-through' : 'none' 
            }}>
              {todo.text}
            </span>
            <button 
              onClick={() => dispatch({ type: 'DELETE_TODO', payload: todo.id })}
            >
              删除
            </button>
          </li>
        ))}
      </ul>
      
      <p>任务总数：{todos.length}，已完成：{todos.filter(t => t.completed).length}</p>
    </div>
  );
}

跨组件状态

使用 Context API 在组件树中共享状态。

// 基础 Context 使用
const ThemeContext = createContext();

function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');
  
  const toggleTheme = useCallback(() => {
    setTheme(prev => prev === 'light' ? 'dark' : 'light');
  }, []);
  
  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
}

// 使用示例
function App() {
  return (
    <ThemeProvider>
      <Header />
      <Main />
      <Footer />
    </ThemeProvider>
  );
}

function Header() {
  const { theme, toggleTheme } = useTheme();
  
  return (
    <header className={theme}>
      <h1>我的应用</h1>
      <button onClick={toggleTheme}>
        当前主题：{theme}
      </button>
    </header>
  );
}

Context 优化技巧

拆分 Context

将不同的状态拆分到不同的 Context 中，避免不必要的重渲染。

// ❌ 不佳实践：所有状态放在一个 Context
const AppContext = createContext();

function AppProvider({ children }) {
  const [theme, setTheme] = useState('light');
  const [user, setUser] = useState(null);
  const [notifications, setNotifications] = useState([]);
  const [settings, setSettings] = useState({});
  
  return (
    <AppContext.Provider value={{
      theme, setTheme,
      user, setUser,
      notifications, setNotifications,
      settings, setSettings
    }}>
      {children}
    </AppContext.Provider>
  );
}

// ✅ 最佳实践：拆分 Context
const ThemeContext = createContext();
const UserContext = createContext();
const NotificationContext = createContext();
const SettingsContext = createContext();

function AppProvider({ children }) {
  return (
    <ThemeProvider>
      <UserProvider>
        <NotificationProvider>
          <SettingsProvider>
            {children}
          </SettingsProvider>
        </NotificationProvider>
      </UserProvider>
    </ThemeProvider>
  );
}

function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');
  
  const value = useMemo(() => ({
    theme,
    toggleTheme: () => setTheme(prev => prev === 'light' ? 'dark' : 'light')
  }), [theme]);
  
  return (
    <ThemeContext.Provider value={value}>
      {children}
    </ThemeContext.Provider>
  );
}

function UserProvider({ children }) {
  const [user, setUser] = useState(null);
  
  const value = useMemo(() => ({
    user,
    login: (userData) => setUser(userData),
    logout: () => setUser(null)
  }), [user]);
  
  return (
    <UserContext.Provider value={value}>
      {children}
    </UserContext.Provider>
  );
}

useMemo 优化 Provider value

function UserProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  
  const login = useCallback(async (credentials) => {
    setLoading(true);
    setError(null);
    
    try {
      const response = await fetch('/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(credentials)
      });
      
      if (!response.ok) {
        throw new Error('登录失败');
      }
      
      const userData = await response.json();
      setUser(userData);
      return userData;
    } catch (err) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, []);
  
  const logout = useCallback(async () => {
    try {
      await fetch('/api/logout', { method: 'POST' });
      setUser(null);
    } catch (err) {
      console.error('登出失败:', err);
    }
  }, []);
  
  const value = useMemo(() => ({
    user,
    loading,
    error,
    login,
    logout
  }), [user, loading, error, login, logout]);
  
  return (
    <UserContext.Provider value={value}>
      {children}
    </UserContext.Provider>
  );
}

import { create } from 'zustand';
import { devtools, persist, subscribeWithSelector } from 'zustand/middleware';

// 基础 Store
const useCounterStore = create((set, get) => ({
  count: 0,
  
  // 方法
  increment: () => set((state) => ({ count: state.count + 1 })),
  decrement: () => set((state) => ({ count: state.count - 1 })),
  reset: () => set({ count: 0 }),
  
  // 异步方法
  incrementAsync: async () => {
    set({ isLoading: true });
    await new Promise(resolve => setTimeout(resolve, 1000));
    set((state) => ({ 
      count: state.count + 1, 
      isLoading: false 
    }));
  },
  
  // 获取其他状态
  getDouble: () => get().count * 2
}));

// 使用 devtools 中间件
const useTodoStore = create(
  devtools(
    (set, get) => ({
      todos: [],
      filter: 'all',
      
      addTodo: (text) => set((state) => ({
        todos: [...state.todos, {
          id: Date.now(),
          text,
          completed: false,
          createdAt: new Date().toISOString()
        }]
      }), false, 'addTodo'),
      
      toggleTodo: (id) => set((state) => ({
        todos: state.todos.map(todo =>
          todo.id === id
            ? { ...todo, completed: !todo.completed }
            : todo
        )
      }), false, 'toggleTodo'),
      
      deleteTodo: (id) => set((state) => ({
        todos: state.todos.filter(todo => todo.id !== id)
      }), false, 'deleteTodo'),
      
      setFilter: (filter) => set({ filter }, false, 'setFilter'),
      
      // 计算属性
      getFilteredTodos: () => {
        const { todos, filter } = get();
        
        switch (filter) {
          case 'completed':
            return todos.filter(todo => todo.completed);
          case 'active':
            return todos.filter(todo => !todo.completed);
          default:
            return todos;
        }
      },
      
      // 统计信息
      getStats: () => {
        const { todos } = get();
        return {
          total: todos.length,
          completed: todos.filter(t => t.completed).length,
          active: todos.filter(t => !t.completed).length
        };
      }
    }),
    { name: 'TodoStore' }
  )
);

// 使用 persist 中间件
const useUserStore = create(
  persist(
    (set, get) => ({
      user: null,
      token: null,

中间件使用

import { create } from 'zustand';
import { 
  devtools, 
  persist, 
  subscribeWithSelector,
  immer 
} from 'zustand/middleware';

// 完整的中间件配置
const useAppStore = create(
  devtools(
    persist(
      subscribeWithSelector(
        immer((set, get) => ({
          // 状态
          user: null,
          theme: 'light',
          notifications: [],
          settings: {
            language: 'zh-CN',
            autoSave: true
          },
          
          // 方法
          setUser: (user) => set((state) => {
            state.user = user;
          }),
          
          setTheme: (theme) => set((state) => {
            state.theme = theme;
          }),
          
          addNotification: (notification) => set((state) => {
            state.notifications.push({
              id: Date.now(),
              timestamp: new Date().toISOString(),
              read: false,
              ...notification
            });
          }),
          
          markNotificationAsRead: (id) => set((state) => {
            const notification = state.notifications.find(n => n.id === id);
            if (notification) {
              notification.read = true;
            }
          }),
          
          updateSettings: (updates) => set((state) => {
            Object.assign(state.settings, updates);
          }),
          
          // 计算属性
          getUnreadCount: () => {
            return get().notifications.filter(n => !n.read).length;
          }
        }))
      ),
      {
        name: 'app-storage',
        partialize: (state) => ({
          theme: state.theme,
          settings: state.settings
        })
      }
    ),
    { name: 'AppStore' }
  )
);

// 订阅状态变化
useAppStore.subscribe(
  (state) => state.theme,
  (theme) => {
    console.log('主题变化:', theme);
    document.documentElement.setAttribute('data-theme', theme);
  }
);

// 使用示例
function ThemeToggle() {

    <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
      当前主题：{theme}
    </button>
  );
}

异步原子

import { atom, useAtom, useAtomValue } from 'jotai';
import { loadable } from 'jotai/utils';

// 异步数据获取 atom
const postsAtom = atom(async (get) => {
  const response = await fetch('/api/posts');
  if (!response.ok) {
    throw new Error('获取文章失败');
  }
  return response.json();
});

// 使用 loadable 包装异步 atom
const postsLoadableAtom = loadable(postsAtom);

// 搜索 atom
const searchTermAtom = atom('');
const filteredPostsAtom = atom(async (get) => {
  const posts = await get(postsAtom);
  const searchTerm = get(searchTermAtom);
  
  if (!searchTerm) return posts;
  
  return posts.filter(post =>
    post.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
    post.content.toLowerCase().includes(searchTerm.toLowerCase())
  );
});

// 分页 atom
const currentPageAtom = atom(1);
const pageSizeAtom = atom(10);
const paginatedPostsAtom = atom(async (get) => {
  const posts = await get(filteredPostsAtom);
  const currentPage = get(currentPageAtom);
  const pageSize = get(pageSizeAtom);
  
  const startIndex = (currentPage - 1) * pageSize;
  const endIndex = startIndex + pageSize;
  
  return {
    posts: posts.slice(startIndex, endIndex),
    totalPages: Math.ceil(posts.length / pageSize),
    currentPage,
    totalPosts: posts.length
  };
});

// 使用示例
function PostList() {
  const postsLoadable = useAtomValue(postsLoadableAtom);
  const [searchTerm, setSearchTerm] = useAtom(searchTermAtom);
  const [currentPage, setCurrentPage] = useAtom(currentPageAtom);
  
  if (postsLoadable.state === 'loading') {
    return <div>加载中...</div>;
  }
  
  if (postsLoadable.state === 'hasError') {
    return <div>错误：{postsLoadable.error.message}</div>;
  }
  
  return (
    <div>
      <input
        type="text"
        value={searchTerm}
        onChange={(e) => setSearchTerm(e.target.value)}
        placeholder="搜索文章..."
      />
      
      <div className="posts">
        {postsLoadable.data.map(post => (
          <article key={post.id} className="post">
            <h3>{post.title}</h3>
            <p>{post.content}</p>
          </article>
        ))}
      </div>
      
      <div className="pagination">
        <button 
          onClick={() => setCurrentPage(p => p - 1)}
          disabled={currentPage === 1}
        >
          上一页
        </button>
        <span>第 {currentPage} 页</span>
        <button onClick={() => setCurrentPage(p => p + 1)}>
          下一页
        </button>
      </div>
    </div>
  );
}

Jotai 与 TanStack Query 集成

import { atom, useAtomValue } from 'jotai';
import { atomsWithQuery, atomsWithMutation } from 'jotai-tanstack-query';

// 查询 atoms
const [userAtom] = atomsWithQuery(() => ({
  queryKey: ['user'],
  queryFn: async () => {
    const response = await fetch('/api/user');
    return response.json();
  }
}));

const [postsAtom] = atomsWithQuery((get) => ({
  queryKey: ['posts', get(searchTermAtom)],
  queryFn: async () => {
    const searchTerm = get(searchTermAtom);
    const response = await fetch(`/api/posts?search=${searchTerm}`);
    return response.json();
  },
  enabled: true
}));

// 变异 atoms
const [createPostAtom] = atomsWithMutation(() => ({
  mutationKey: ['createPost'],
  mutationFn: async (newPost) => {
    const response = await fetch('/api/posts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(newPost)
    });
    return response.json();
  }
}));

// 使用示例
function CreatePostForm() {
  const [createPost, mutation] = useAtom(createPostAtom);
  const [title, setTitle] = useState('');
  const [content, setContent] = useState('');
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    try {
      await createPost({ title, content });
      setTitle('');
      setContent('');
      alert('发布成功！');
    } catch (error) {
      alert('发布失败：' + error.message);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        placeholder="标题"
        required
      />
      <textarea
        value={content}
        onChange={(e) => setContent(e.target.value)}
        placeholder="内容"
        required
      />
      <button type="submit" disabled={mutation.isLoading}>
        {mutation.isLoading ? '发布中...' : '发布'}
      </button>
      
      {mutation.isError && (
        <p className="error">错误：{mutation.error.message}</p>
      )}
    </form>
  );
}

代码示例：用 Zustand 实现完整的全局状态管理

import { create } from 'zustand';
import { devtools, persist, subscribeWithSelector } from 'zustand/middleware';

// 用户状态管理
const useUserStore = create(
  devtools(
    persist(
      (set, get) => ({
        user: null,
        token: null,
        isLoading: false,
        error: null,
        
        // 登录
        login: async (credentials) => {
          set({ isLoading: true, error: null });
          
          try {
            const response = await fetch('/api/login', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(credentials)
            });
            
            if (!response.ok) {
              throw new Error('登录失败');
            }
            
            const { user, token } = await response.json();
            set({ user, token, isLoading: false });
            
            return user;
          } catch (error) {
            set({ error: error.message, isLoading: false });
            throw error;
          }
        },
        
        // 登出
        logout: async () => {
          try {
            await fetch('/api/logout', { method: 'POST' });
          } finally {
            set({ user: null, token: null });
          }
        },
        
        // 更新用户信息
        updateProfile: async (updates) => {
          set({ isLoading: true });
          
          try {
            const response = await fetch('/api/user/profile', {
              method: 'PATCH',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(updates)
            });
            
            if (!response.ok) {
              throw new Error('更新失败');
            }
            
            const updatedUser = await response.json();
            set((state) => ({
              user: { ...state.user, ...updatedUser },
              isLoading: false
            }));
          } catch (error) {
            set({ isLoading: false });
            throw error;
          }
        },
        
        // 检查登录状态
        checkAuth: async () => {
          const token = get().token;
          if (!token) return;
          
          try {
            const response = await fetch('/api/auth/check', {
              headers: { Authorization: `Bearer ${token}` }
            });
            
            if (!response.ok) {
              throw new Error('Token 无效');
            }
            
            const user = await response.json();
            set({ user });
          } catch (error) {
            set({ user: null, token: null });
          }
        }
      }),
      {
        name: 'user-storage',
        partialize: (state) => ({
          token: state.token
        })
      }
    ),
    { name: 'UserStore' }
  )
);

// 购物车状态管理
const useCartStore = create(
  devtools(
    (set, get) => ({
      items: [],
      coupon: null,
      
      // 添加商品
      addItem: (product, quantity = 1) => set((state) => {
        const existingItem = state.items.find(
          item => item.productId === product.id
        );
        
        if (existingItem) {
          return {
            items: state.items.map(item =>
              item.productId === product.id
                ? { ...item, quantity: item.quantity + quantity }
                : item
            )
          };
        }
        
        return {
          items: [...state.items, {
            productId: product.id,
            name: product.name,
            price: product.price,
            image: product.image,
            quantity
          }]
        };
      }),
      
      // 移除商品
      removeItem: (productId) => set((state) => ({
        items: state.items.filter(item => item.productId !== productId)
      })),
      
      // 更新数量
      updateQuantity: (productId, quantity) => set((state) => ({
        items: quantity <= 0
          ? state.items.filter(item => item.productId !== productId)
          : state.items.map(item =>
              item.productId === productId
                ? { ...item, quantity }
                : item
            )
      })),
      
      // 清空购物车
      clearCart: () => set({ items: [], coupon: null }),
      
      // 使用优惠券
      applyCoupon: (coupon) => {
        const { getSubtotal } = get();
        const subtotal = getSubtotal();
        
        if (subtotal < coupon.minAmount) {
          throw new Error(`订单金额需满 ${coupon.minAmount} 元`);
        }
        
        set({ coupon });
      },
      
      // 移除优惠券
      removeCoupon: () => set({ coupon: null }),
      
      // 计算小计
      getSubtotal: () => {
        const { items } = get();
        return items.reduce((sum, item) => sum + item.price * item.quantity, 0);
      },
      
      // 计算折扣
      getDiscount: () => {
        const { coupon } = get();
        const subtotal = get().getSubtotal();
        
        if (!coupon) return 0;
        
        if (coupon.type === 'percentage') {
          return subtotal * (coupon.value / 100);
        }
        
        return Math.min(coupon.value, subtotal);
      },
      
      // 计算总价
      getTotal: () => {
        const subtotal = get().getSubtotal();
        const discount = get().getDiscount();
        const shipping = subtotal >= 99 ? 0 : 10;
        
        return subtotal - discount + shipping;
      },
      
      // 获取商品总数
      getItemCount: () => {
        const { items } = get();
        return items.reduce((sum, item) => sum + item.quantity, 0);
      }
    }),
    { name: 'CartStore' }
  )
);

// 通知状态管理
const useNotificationStore = create(
  devtools(
    (set, get) => ({
      notifications: [],
      
      // 添加通知
      addNotification: (notification) => {
        const id = Date.now();
        const newNotification = {
          id,
          timestamp: new Date().toISOString(),
          read: false,
          ...notification
        };
        
        set((state) => ({
          notifications: [newNotification, ...state.notifications]
        }));
        
        // 自动移除通知
        if (notification.duration !== 0) {
          setTimeout(() => {
            get().removeNotification(id);
          }, notification.duration || 5000);
        }
        
        return id;
      },
      
      // 移除通知
      removeNotification: (id) => set((state) => ({
        notifications: state.notifications.filter(n => n.id !== id)
      })),
      
      // 标记为已读
      markAsRead: (id) => set((state) => ({
        notifications: state.notifications.map(n =>
          n.id === id ? { ...n, read: true } : n
        )
      })),
      
      // 清除所有通知
      clearAll: () => set({ notifications: [] }),
      
      // 获取未读数量
      getUnreadCount: () => {
        return get().notifications.filter(n => !n.read).length;
      }
    }),
    { name: 'NotificationStore' }
  )
);

// 使用示例
function App() {
  const { user, checkAuth } = useUserStore();
  const { getItemCount } = useCartStore();
  const { getUnreadCount } = useNotificationStore();
  
  useEffect(() => {
    checkAuth();
  }, [checkAuth]);
  
  return (
    <div className="app">
      <Header 
        user={user}
        cartItemCount={getItemCount()}
        unreadNotifications={getUnreadCount()}
      />
      <Main />
      <Footer />
    </div>
  );
}

function Header({ user, cartItemCount, unreadNotifications }) {
  const { logout } = useUserStore();
  const { addNotification } = useNotificationStore();
  
  const handleLogout = async () => {
    try {
      await logout();
      addNotification({
        type: 'success',
        message: '已成功登出'
      });
    } catch (error) {
      addNotification({
        type: 'error',
        message: '登出失败'
      });
    }
  };
  
  return (
    <header>
      <div className="logo">我的商城</div>
      
      <nav>
        <a href="/products">商品</a>
        <a href="/cart">
          购物车 {cartItemCount > 0 && (
            <span className="badge">{cartItemCount}</span>
          )}
        </a>
      </nav>
      
      <div className="user-menu">
        {unreadNotifications > 0 && (
          <span className="notification-badge">{unreadNotifications}</span>
        )}
        
        {user ? (
          <div className="user-info">
            <span>{user.name}</span>
            <button onClick={handleLogout}>登出</button>
          </div>
        ) : (
          <a href="/login">登录</a>
        )}
      </div>
    </header>
  );
}

export { useUserStore, useCartStore, useNotificationStore };

要点总结

  1. 状态层次：组件内 → 跨组件 → 全局，逐层升级
  2. Context 优化：拆分 Context、使用 useMemo 优化 Provider value
  3. Zustand：轻量、简洁，适合中小型应用
  4. Jotai：原子化状态，适合复杂的状态依赖关系
  5. Redux Toolkit：功能强大，适合大型应用，但模板代码较多
  6. 选型建议：根据项目规模、团队熟悉度、性能需求选择
  7. 最佳实践：避免过度设计，从简单方案开始，必要时升级
