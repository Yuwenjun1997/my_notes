流程图编辑器：React Flow 实战

约 1097 字大约 4 分钟

节点编辑器概述

节点编辑器（Node-based Editor）是一种用"节点 + 连线"来表达流程、逻辑或数据关系的可视化工具。在低代码平台中，常用于：

  • 审批流程设计：配置多级审批、条件分支
  • 数据处理管线：ETL 流程可视化编排
  • 逻辑编排：用节点组合替代 if-else 代码
  • AI 工作流：类似 LangChain 的 Agent 编排

React Flow 是 React 生态中最成熟的节点编辑器库，支持自定义节点、多种连线类型、迷你地图、minimap 等。

React Flow 核心概念

基础结构

import ReactFlow, { MiniMap, Controls, Background } from 'reactflow';
import 'reactflow/dist/style.css';

const initialNodes = [
  { id: '1', position: { x: 0, y: 0 }, data: { label: '开始' }, type: 'input' },
  { id: '2', position: { x: 200, y: 0 }, data: { label: '处理数据' } },
  { id: '3', position: { x: 400, y: 0 }, data: { label: '结束' }, type: 'output' },
];

const initialEdges = [
  { id: 'e1-2', source: '1', target: '2', animated: true },
  { id: 'e2-3', source: '2', target: '3' },
];

function FlowEditor() {
  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges);

  const onConnect = useCallback((params) => {
    setEdges(eds => addEdge(params, eds));
  }, []);

  return (
    <ReactFlow
      nodes={nodes}
      edges={edges}
      onNodesChange={onNodesChange}
      onEdgesChange={onEdgesChange}
      onConnect={onConnect}
    >
      <MiniMap />
      <Controls />
      <Background />
    </ReactFlow>
  );
}

React Flow 采用受控模式：节点和边的数据由外部 state 管理，通过 onNodesChange / onEdgesChange 回调更新。

节点和边

  • Node：{ id, position: { x, y }, data: { label, ... }, type } — type 决定渲染哪个组件
  • Edge：{ id, source, target, type, animated, label } — 连接两个节点

自定义节点开发

为流程编辑器创建四种常用节点：

// 1. 开始节点
function StartNode({ data }) {
  return (
    <div style={{ padding: '8px 16px', borderRadius: 20, background: '#52c41a', color: '#fff' }}>
      {data.label}
      <Handle type="source" position={Position.Right} />
    </div>
  );
}

// 2. 处理节点（带表单属性）
function ActionNode({ data }) {
  return (
    <div style={{ padding: 12, border: '1px solid #d9d9d9', borderRadius: 8, background: '#fff' }}>
      <Handle type="target" position={Position.Left} />
      <div style={{ fontWeight: 'bold', marginBottom: 4 }}>{data.label}</div>
      <div style={{ fontSize: 12, color: '#666' }}>{data.description || '点击配置操作'}</div>
      <Handle type="source" position={Position.Right} />
    </div>
  );
}

// 3. 条件节点（带两个出口）
function ConditionNode({ data }) {
  return (
    <div style={{ padding: 12, border: '2px solid #faad14', borderRadius: 8, background: '#fffbe6' }}>
      <Handle type="target" position={Position.Left} />
      <div style={{ fontWeight: 'bold', marginBottom: 8 }}>{data.label}</div>
      <div style={{ position: 'relative', height: 20 }}>
        <Handle
          type="source"
          position={Position.Right}
          id="true"
          style={{ top: -10, background: '#52c41a' }}
        />
        <Handle
          type="source"
          position={Position.Right}
          id="false"
          style={{ top: 10, background: '#f5222d' }}
        />
      </div>
      <div style={{ fontSize: 11, display: 'flex', justifyContent: 'space-between' }}>
        <span style={{ color: '#52c41a' }}>是</span>
        <span style={{ color: '#f5222d' }}>否</span>
      </div>
    </div>
  );
}

// 4. 结束节点
function EndNode({ data }) {
  return (
    <div style={{ padding: '8px 16px', borderRadius: 20, background: '#f5222d', color: '#fff' }}>
      <Handle type="target" position={Position.Left} />
      {data.label}
    </div>
  );
}

// 注册自定义节点类型
const nodeTypes = {
  start: StartNode,
  action: ActionNode,
  condition: ConditionNode,
  end: EndNode,
};

// 使用
<ReactFlow nodeTypes={nodeTypes} nodes={nodes} edges={edges} ... />

连线与交互

连线验证

不是任意两个节点都能连线，需要验证：

function FlowEditor() {
  const onConnect = useCallback((params) => {
    // 规则：条件节点只能连接到 action 或 end 节点
    const sourceNode = nodes.find(n => n.id === params.source);
    const targetNode = nodes.find(n => n.id === params.target);

    if (sourceNode?.type === 'condition' && targetNode?.type === 'start') {
      return;  // 拒绝连接
    }
    setEdges(eds => addEdge({ ...params, animated: false }, eds));
  }, [nodes]);

  // 连线验证：拖拽连线时实时判断
  const isValidConnection = useCallback((connection) => {
    return connection.source !== connection.target;  // 不能连自己
  }, []);
}

<ReactFlow isValidConnection={isValidConnection} onConnect={onConnect} ... />

function NodePalette() {
  const nodeTemplates = [
    { type: 'action', label: '处理节点', icon: '⚙️' },
    { type: 'condition', label: '条件判断', icon: '❓' },
    { type: 'end', label: '结束', icon: '🛑' },
  ];

  return (
    <div className="node-palette">
      <h4>节点</h4>
      {nodeTemplates.map(t => (
        <div
          key={t.type}
          className="palette-node"
          draggable
          onDragStart={e => e.dataTransfer.setData('nodeType', t.type)}
        >
          {t.icon} {t.label}
        </div>
      ))}
    </div>
  );
}

// 画布接收拖入
function FlowCanvas() {
  const onDragOver = (e) => { e.preventDefault(); };

  const onDrop = (e) => {
    const type = e.dataTransfer.getData('nodeType');
    const position = screenToFlowPosition({ x: e.clientX, y: e.clientY });
    const newNode = {
      id: nanoid(),
      type,
      position,
      data: { label: getDefaultLabel(type) },
    };
    setNodes(nds => [...nds, newNode]);
  };

  return <ReactFlow onDragOver={onDragOver} onDrop={onDrop} ... />;
}

流程序列化

保存和加载流程定义：

// 导出为 JSON
function exportFlow(nodes, edges) {
  return JSON.stringify({ nodes, edges }, null, 2);
}

// 导入
function importFlow(json: string) {
  const { nodes, edges } = JSON.parse(json);
  setNodes(nodes);
  setEdges(edges);
}

// 简单的流程执行：按拓扑顺序遍历节点
async function executeFlow(nodes, edges) {
  const adjacency = buildAdjacencyList(nodes, edges);
  const startNode = nodes.find(n => n.type === 'start');
  const queue = [startNode];

  while (queue.length > 0) {
    const node = queue.shift();
    await executeNode(node);  // 执行节点逻辑

    const nextEdges = adjacency[node.id] || [];
    for (const edge of nextEdges) {
      if (edge.sourceHandle === 'true' || !edge.sourceHandle) {
        queue.push(nodes.find(n => n.id === edge.target));
      }
    }
  }
}
