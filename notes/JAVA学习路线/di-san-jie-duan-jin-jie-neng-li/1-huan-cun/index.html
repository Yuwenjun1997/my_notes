一、Redis 数据结构详解

Redis 是一个高性能的键值存储系统，支持多种数据结构。

1.1 String（字符串）

String 是 Redis 最基本的数据类型，最大存储 512MB。

# 基本操作
SET key "value"              # 设置键值
GET key                      # 获取值
MSET k1 v1 k2 v2             # 批量设置
MGET k1 k2                   # 批量获取
STRLEN key                   # 获取字符串长度

# 数值操作
INCR counter                  # 自增 1（原子操作）
INCRBY counter 10             # 增加 10
DECR counter                  # 自减 1
DECRBY counter 5              # 减少 5

# 过期时间
SET key "value" EX 3600      # 设置 1 小时过期
SETEX key 3600 "value"       # 同上，原子操作
SET key "value" NX           # 仅 key 不存在时设置（用于分布式锁）
TTL key                      # 查看剩余过期时间（秒）
EXPIRE key 3600              # 设置过期时间

# 应用场景
# 1. 缓存数据
SET user:1001:profile '{"name":"张三","age":25}' EX 3600

# 2. 计数器（点赞数、PV、UV 等）
INCR article:1001:views

# 3. 分布式锁
SET lock:order:1001 "uuid-xxx" NX EX 30

Spring Boot 操作 String

@Autowired
private StringRedisTemplate redisTemplate;

// 缓存用户信息
public void cacheUser(User user) {
    String key = "user:" + user.getId() + ":profile";
    String value = JSON.toJSONString(user);
    redisTemplate.opsForValue().set(key, value, 1, TimeUnit.HOURS);
}

// 获取缓存
public User getCachedUser(Long userId) {
    String key = "user:" + userId + ":profile";
    String value = redisTemplate.opsForValue().get(key);
    if (value != null) {
        return JSON.parseObject(value, User.class);
    }
    return null;
}

// 原子计数
public long incrementViews(Long articleId) {
    String key = "article:" + articleId + ":views";
    return redisTemplate.opsForValue().increment(key);
}

1.2 List（列表）

List 是基于双向链表实现的，支持从两端插入和弹出。

# 从右侧插入（队列：FIFO）
RPUSH queue "task1"          # 从右侧推入
RPUSH queue "task2" "task3"  # 批量推入
LPOP queue                   # 从左侧弹出 → "task1"

# 从左侧插入（栈：LIFO）
LPUSH stack "a"               # 从左侧推入
RPOP stack                    # 从右侧弹出

# 阻塞操作（常用于消息队列）
BLPOP queue 0                # 阻塞从左侧弹出，0 表示无限等待
BRPOP queue 30               # 阻塞从右侧弹出，超时 30 秒

# 范围查询
LRANGE list 0 -1             # 获取所有元素
LLEN list                    # 获取列表长度
LINDEX list 0                # 获取索引为 0 的元素

应用场景：消息队列

// 生产者
public void sendTask(String task) {
    redisTemplate.opsForList().rightPush("task:queue", task);
}

// 消费者（阻塞获取）
public void consumeTask() {
    while (true) {
        // BLPOP：阻塞式弹出，0 表示无限等待
        String task = redisTemplate.opsForList()
            .leftPop("task:queue", 0, TimeUnit.SECONDS);
        processTask(task);
    }
}

1.3 Set（集合）

Set 是无序且元素唯一的字符串集合。

# 基本操作
SADD tags "java" "spring"    # 添加元素（重复元素自动忽略）
SMEMBERS tags                 # 获取所有元素
SISMEMBER tags "java"         # 判断元素是否存在 → 1
SCARD tags                    # 获取元素数量
SREM tags "java"              # 删除元素

# 集合运算（强大！）
SINTER set1 set2              # 交集
SUNION set1 set2              # 并集
SDIFF set1 set2               # 差集（在 set1 但不在 set2 中的元素）

应用场景：好友关系、标签系统

// 给文章打标签
public void addArticleTags(Long articleId, List<String> tags) {
    String key = "article:" + articleId + ":tags";
    redisTemplate.opsForSet().add(key, tags.toArray());
}

// 查找同时拥有 "java" 和 "spring" 标签的文章
Set<String> intersect = redisTemplate.opsForSet()
    .intersect("tag:java:articles", "tag:spring:articles");

1.4 Hash（哈希）

Hash 用于存储对象结构，类似于 Java 的 Map<String, String>。

# 基本操作
HSET user:1001 name "张三"     # 设置单个字段
HSET user:1001 age 25 email "z@e.com"  # 设置多个字段
HGET user:1001 name            # 获取单个字段
HMGET user:1001 name age       # 获取多个字段
HGETALL user:1001              # 获取所有字段
HDEL user:1001 email           # 删除字段
HEXISTS user:1001 name         # 判断字段是否存在

# 数值操作
HINCRBY user:1001 score 10     # 原子增加字段值

应用场景：对象缓存

// Hash 方式缓存对象（适合频繁修改部分字段）
public void cacheUserHash(User user) {
    String key = "user:" + user.getId();
    Map<String, Object> map = new HashMap<>();
    map.put("name", user.getName());
    map.put("age", user.getAge());
    map.put("email", user.getEmail());
    map.put("score", user.getScore());
    redisTemplate.opsForHash().putAll(key, map);
}

// 只更新用户积分（无需反序列化整个对象）
public void updateScore(Long userId, int delta) {
    String key = "user:" + userId;
    redisTemplate.opsForHash().increment(key, "score", delta);
}

1.5 ZSet（有序集合）

ZSet 每个元素关联一个 score（分数），按 score 排序。

# 基本操作
ZADD leaderboard 100 "player1"     # 添加元素并设置分数
ZADD leaderboard 200 "player2" 150 "player3"
ZRANGE leaderboard 0 -1             # 按分数升序获取所有
ZREVRANGE leaderboard 0 -1          # 按分数降序获取所有
ZRANK leaderboard "player1"         # 获取排名（升序，从 0 开始）
ZREVRANK leaderboard "player1"      # 获取排名（降序）
ZSCORE leaderboard "player1"        # 获取分数
ZINCRBY leaderboard 50 "player1"    # 增加分数

# 范围查询
ZREVRANGE leaderboard 0 9 WITHSCORES  # 排行榜前 10 名
ZRANGEBYSCORE leaderboard 100 200     # 分数在 100-200 之间的元素
ZREMRANGEBYRANK leaderboard 0 99      # 删除前 100 名之外的

应用场景：排行榜

// 更新玩家分数
public void updateScore(Long playerId, int score) {
    redisTemplate.opsForZSet()
        .incrementScore("game:leaderboard", String.valueOf(playerId), score);
}

// 获取排行榜前 10 名
public List<PlayerRank> getTop10() {
    Set<ZSetOperations.TypedTuple<String>> top10 = redisTemplate.opsForZSet()
        .reverseRangeWithScores("game:leaderboard", 0, 9);

    List<PlayerRank> ranks = new ArrayList<>();
    int rank = 1;
    for (ZSetOperations.TypedTuple<String> tuple : top10) {
        PlayerRank pr = new PlayerRank();
        pr.setRank(rank++);
        pr.setPlayerId(Long.valueOf(tuple.getValue()));
        pr.setScore(tuple.getScore().intValue());
        ranks.add(pr);
    }
    return ranks;
}

二、Redis 持久化

2.1 RDB（快照持久化）

RDB 通过生成数据集的快照来持久化，按一定间隔将内存数据写入磁盘。

工作原理

Redis fork 出子进程
    ↓
子进程将内存数据写入临时 RDB 文件
    ↓
写入完成后替换旧的 RDB 文件（原子替换）
    ↓
父进程继续处理请求
    ↓
子进程退出

配置方式

# redis.conf
# 自动触发条件（满足任一即触发）
save 900 1          # 900 秒内至少 1 个 key 变化
save 300 10         # 300 秒内至少 10 个 key 变化
save 60 10000       # 60 秒内至少 10000 个 key 变化

# RDB 文件配置
dbfilename dump.rdb
dir /var/lib/redis
stop-writes-on-bgsave-error yes    # 持久化失败时停止写入
rdbcompression yes                  # 压缩 RDB 文件

手动触发

SAVE      # 同步持久化（阻塞所有客户端，生产环境不要用）
BGSAVE    # 异步持久化（后台 fork 子进程执行）
LASTSAVE  # 查看最后一次成功持久化的时间戳

优点：文件紧凑、恢复速度快、适合备份 缺点：会丢失最后一次快照后的数据（数据一致性差）

2.2 AOF（追加文件持久化）

AOF 记录每次写操作命令，重启时重放这些命令。

工作原理

客户端发送写命令
    ↓
命令追加到 AOF 缓冲区
    ↓
根据刷盘策略（appendfsync）将缓冲区写入磁盘
    ↓
AOF 文件过大时触发 AOF 重写（压缩）

配置方式

# redis.conf
appendonly yes                   # 开启 AOF
appendfilename "appendonly.aof"

# 刷盘策略
appendfsync always               # 每次写入都刷盘（最安全，最慢）
appendfsync everysec             # 每秒刷一次（推荐，最多丢 1 秒数据）
appendfsync no                   # 由操作系统决定（最快，丢数据最多）

# AOF 重写
auto-aof-rewrite-percentage 100  # AOF 文件增长超过 100% 时触发重写
auto-aof-rewrite-min-size 64mb   # AOF 文件至少 64MB 才触发重写
no-appendfsync-on-rewrite yes    # 重写期间不执行 fsync，防止阻塞

AOF 重写原理

Redis fork 子进程
    ↓
子进程将当前内存数据转换为最小命令集
    ↓
写入临时 AOF 文件
    ↓
切换为新 AOF 文件

优点：数据安全性高（最多丢 1 秒数据）、可读性好 缺点：文件体积大、恢复速度慢

2.3 混合持久化（Redis 4.0+）

结合 RDB 和 AOF 的优点：

# redis.conf
aof-use-rdb-preamble yes

原理：

  • AOF 重写时：先写入 RDB 格式的快照数据 → 再追加增量 AOF 数据
  • 重启恢复时：先加载 RDB（快速）→ 再重放 AOF（增量）

RDB vs AOF vs 混合

特性RDBAOF混合（默认推荐）
数据安全性低（可能丢大量数据）高（最多丢 1s）高
恢复速度快慢快
文件大小小大适中
资源消耗fork 子进程持续写入适中
适用场景备份、灾难恢复数据安全要求高默认推荐

三、缓存穿透、击穿、雪崩

3.1 缓存穿透

定义：查询一个不存在的数据，请求直接打到数据库。

正常流程：
请求 → 查询缓存 → 命中 → 返回
请求 → 查询缓存 → 未命中 → 查询数据库 → 回写缓存 → 返回

缓存穿透：
请求 → 查询缓存 → 未命中（数据不存在） → 查询数据库 → 不存在 → 返回空
                                                          ← 每次请求都查数据库

解决方案

// 方案一：缓存空对象（推荐）
public User getUserById(Long id) {
    String key = "user:" + id;

    // 1. 查缓存
    User user = cacheService.get(key);
    if (user != null) {
        return user;  // 缓存命中（包括空值）
    }

    // 2. 查数据库
    user = userMapper.findById(id);

    // 3. 缓存结果（即使是 null 也缓存，过期时间短一些）
    if (user == null) {
        cacheService.set(key, new User(), 60);  // 空值缓存 60 秒
    } else {
        cacheService.set(key, user, 3600);
    }

    return user;
}

// 方案二：布隆过滤器（Bloom Filter）
// 原理：用多个哈希函数判断元素"不存在"或"可能存在"
@Component
public class BloomFilterService {

    private BloomFilter<Long> bloomFilter;

    @PostConstruct
    public void init() {
        // 预计插入 10000 个元素，误判率 1%
        bloomFilter = BloomFilter.create(Funnels.longFunnel(), 10000, 0.01);

        // 加载所有存在的 ID 到布隆过滤器
        List<Long> allIds = userMapper.findAllIds();
        allIds.forEach(bloomFilter::put);
    }

    public boolean mightContain(Long id) {
        return bloomFilter.mightContain(id);
    }
}

// 在查询中使用
public User getUserById(Long id) {
    // 布隆过滤器判断不存在 → 直接返回
    if (!bloomFilterService.mightContain(id)) {
        return null;  // 肯定不存在
    }

    // 可能存在的，继续查缓存 → 数据库
    // ...
}

3.2 缓存击穿

定义：高并发访问某个热点 key，该 key 刚好失效，大量请求直接打到数据库。

正常情况：
高并发请求 → 缓存中有数据 → 全部返回

缓存击穿：
高并发请求 → 缓存刚好过期
            ↓
请求1 → 查数据库（获取锁）
请求2 → 查数据库（发现锁被占用，等待）
请求3 → 查数据库（等待）
            ↓
请求1 回写缓存 → 释放锁
请求2 → 查缓存命中 → 返回
请求3 → 查缓存命中 → 返回

解决方案

// 方案一：互斥锁（Mutex Key）
public String getData(String key) {
    // 1. 尝试从缓存获取
    String data = redisTemplate.opsForValue().get(key);

    if (data == null) {
        // 2. 获取分布式锁
        String lockKey = "lock:" + key;
        boolean locked = redisTemplate.opsForValue()
                .setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);

        if (locked) {
            // 3. 再次检查缓存（double check）
            data = redisTemplate.opsForValue().get(key);
            if (data != null) {
                redisTemplate.delete(lockKey);
                return data;
            }

            // 4. 查数据库
            data = queryFromDB(key);

            // 5. 回写缓存
            redisTemplate.opsForValue().set(key, data, 3600, TimeUnit.SECONDS);

            // 6. 释放锁
            redisTemplate.delete(lockKey);
        } else {
            // 7. 没获取到锁，等待重试
            Thread.sleep(100);
            return getData(key);  // 递归重试
        }
    }

    return data;
}

// 方案二：热点数据永不过期（逻辑过期）
// 缓存不设置物理过期时间，而是存一个逻辑过期时间
public String getDataWithLogicalExpire(String key) {
    // 1. 查缓存
    CacheData cached = redisTemplate.opsForValue().get(key);

    if (cached == null) {
        // 缓存不存在，查数据库
        return queryFromDB(key);
    }

    // 2. 判断是否逻辑过期
    if (cached.getExpireTime() > System.currentTimeMillis()) {
        // 未过期，直接返回
        return cached.getData();
    }

    // 3. 已过期，异步更新缓存（获取锁后由单个线程更新）
    String lockKey = "lock:" + key;
    if (redisTemplate.opsForValue().setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS)) {
        // 异步线程更新缓存
        CompletableFuture.runAsync(() -> {
            String newData = queryFromDB(key);
            CacheData newCached = new CacheData(newData, System.currentTimeMillis() + 3600_000);
            redisTemplate.opsForValue().set(key, newCached);
            redisTemplate.delete(lockKey);
        });
    }

    // 4. 先返回旧数据（保证可用性）
    return cached.getData();
}

3.3 缓存雪崩

定义：大量缓存同时过期，或 Redis 宕机，导致请求全部打到数据库。

缓存雪崩原因：
1. 大量 key 设置了相同的过期时间
2. Redis 实例宕机
3. 缓存服务重启

结果：数据库连接数暴增，可能被打垮

解决方案

// 方案一：过期时间加随机值
public void cacheUser(User user) {
    String key = "user:" + user.getId();
    // 基础时间 1 小时 + 随机 0-300 秒
    int timeout = 3600 + new Random().nextInt(300);
    redisTemplate.opsForValue().set(key, user, timeout, TimeUnit.SECONDS);
}

// 方案二：多级缓存（本地缓存 + Redis）
// 使用 Caffeine 作为一级缓存，Redis 作为二级缓存
@Component
public class MultiLevelCache {

    // 一级缓存：Caffeine（本地内存）
    private final Cache<String, Object> localCache = Caffeine.newBuilder()
            .maximumSize(10000)
            .expireAfterWrite(10, TimeUnit.MINUTES)
            .build();

    // 二级缓存：Redis
    @Autowired
    private StringRedisTemplate redisTemplate;

    public Object get(String key) {
        // 1. 查一级缓存
        Object value = localCache.getIfPresent(key);
        if (value != null) {
            return value;
        }

        // 2. 查二级缓存
        String json = redisTemplate.opsForValue().get(key);
        if (json != null) {
            // 回填一级缓存
            localCache.put(key, json);
            return json;
        }

        return null;  // 都未命中，查数据库
    }

    public void put(String key, Object value, int timeout, TimeUnit unit) {
        localCache.put(key, value);
        redisTemplate.opsForValue().set(key, value.toString(), timeout, unit);
    }
}

// 方案三：Redis 高可用
// 1. 主从复制 + 哨兵
// 2. Redis Cluster
// 3. 持久化策略确保重启后快速恢复

// 方案四：限流降级
// 1. 使用 Hystrix/Sentinel 进行服务降级
// 2. 限制对数据库的并发访问

三种问题对比总结

问题原因表现解决方案
缓存穿透查询不存在的数据缓存+数据库都查不到空值缓存、布隆过滤器
缓存击穿热点 key 失效大量并发查数据库互斥锁、逻辑过期
缓存雪崩大量 key 同时失效数据库被打垮过期时间加随机值、多级缓存、高可用

四、分布式锁

4.1 基于 Redis 的分布式锁

基本实现

@Component
public class RedisDistributedLock {

    @Autowired
    private StringRedisTemplate redisTemplate;

    /**
     * 获取分布式锁
     * @param lockKey   锁键
     * @param requestId 请求标识（用于解锁时验证，防止误删别人的锁）
     * @param expireTime 过期时间（毫秒）
     * @return true 获取成功
     */
    public boolean tryLock(String lockKey, String requestId, long expireTime) {
        Boolean result = redisTemplate.opsForValue()
                .setIfAbsent(lockKey, requestId, expireTime, TimeUnit.MILLISECONDS);
        return Boolean.TRUE.equals(result);
    }

    /**
     * 释放锁（使用 Lua 脚本保证原子性）
     */
    public boolean unlock(String lockKey, String requestId) {
        // Lua 脚本：先判断是否自己的锁，再删除
        String script = "if redis.call('get', KEYS[1]) == ARGV[1] " +
                        "then return redis.call('del', KEYS[1]) " +
                        "else return 0 end";

        DefaultRedisScript<Long> redisScript = new DefaultRedisScript<>();
        redisScript.setScriptText(script);
        redisScript.setResultType(Long.class);

        Long result = redisTemplate.execute(redisScript,
                Collections.singletonList(lockKey), requestId);
        return Long.valueOf(1).equals(result);
    }
}

使用示例

@Service
public class OrderService {

    @Autowired
    private RedisDistributedLock redisLock;

    public boolean createOrder(Long orderId) {
        String lockKey = "lock:order:" + orderId;
        String requestId = UUID.randomUUID().toString();

        try {
            // 尝试获取锁
            boolean locked = redisLock.tryLock(lockKey, requestId, 30000);
            if (!locked) {
                throw new RuntimeException("订单正在处理中");
            }

            // 业务处理（保证幂等性）
            return doCreateOrder(orderId);

        } finally {
            // 释放锁
            redisLock.unlock(lockKey, requestId);
        }
    }
}

4.2 Redisson 实现（推荐）

Redisson 提供了更完善的分布式锁实现（自动续期、可重入等）。

<dependency>
    <groupId>org.redisson</groupId>
    <artifactId>redisson-spring-boot-starter</artifactId>
    <version>3.25.0</version>
</dependency>
@Configuration
public class RedissonConfig {

    @Bean
    public RedissonClient redissonClient() {
        Config config = new Config();
        config.useSingleServer()
                .setAddress("redis://localhost:6379")
                .setPassword(null)
                .setConnectionPoolSize(10);
        return Redisson.create(config);
    }
}

@Service
public class OrderService {

    @Autowired
    private RedissonClient redissonClient;

    public boolean createOrder(Long orderId) {
        RLock lock = redissonClient.getLock("lock:order:" + orderId);

        try {
            // 尝试加锁，最多等待 10 秒，锁过期自动释放 30 秒
            if (lock.tryLock(10, 30, TimeUnit.SECONDS)) {
                try {
                    // 业务处理
                    return doCreateOrder(orderId);
                } finally {
                    if (lock.isHeldByCurrentThread()) {
                        lock.unlock();  // 释放锁
                    }
                }
            } else {
                throw new RuntimeException("获取锁失败，请稍后重试");
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new RuntimeException("获取锁被中断", e);
        }
    }
}

4.3 分布式锁对比

方案优点缺点适用场景
Redis 自实现轻量、可控需处理边界情况（续期、可重入）简单场景
Redisson功能完善（看门狗、可重入）增加依赖大多数场景
ZooKeeper强一致性、自动释放性能不如 Redis对一致性要求极高的场景
数据库简单、无需额外组件性能差几乎不用
