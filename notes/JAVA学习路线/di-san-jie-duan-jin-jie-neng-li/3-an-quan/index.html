一、Spring Security

1.1 核心概念

Spring Security 是 Spring 生态中的安全框架，提供认证（Authentication）和授权（Authorization）功能。

核心过滤器链

请求到达 → 安全过滤器链（SecurityFilterChain）
    ↓
SecurityContextPersistenceFilter     ← 从 Session 中加载 SecurityContext
    ↓
LogoutFilter                         ← 处理登出
    ↓
UsernamePasswordAuthenticationFilter ← 处理表单登录（用户名密码认证）
    ↓
BasicAuthenticationFilter            ← HTTP Basic 认证
    ↓
RequestCacheAwareFilter              ← 缓存请求（登录后跳回原页面）
    ↓
ExceptionTranslationFilter           ← 转换异常为 HTTP 响应
    ↓
FilterSecurityInterceptor            ← 最终授权决策
    ↓
访问 Controller

Security 配置类

@Configuration
@EnableWebSecurity
@EnableMethodSecurity  // 开启方法级别安全注解
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
            // 禁用 CSRF（前后端分离项目）
            .csrf(AbstractHttpConfigurer::disable)

            // 无状态（使用 JWT，不创建 Session）
            .sessionManagement(session ->
                session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))

            // 配置 URL 权限
            .authorizeHttpRequests(auth -> auth
                // 公开接口（无需认证）
                .requestMatchers("/api/auth/**").permitAll()
                .requestMatchers("/api/public/**").permitAll()
                .requestMatchers("/swagger-ui/**", "/v3/api-docs/**").permitAll()
                .requestMatchers("/actuator/health").permitAll()
                // 管理员接口
                .requestMatchers("/api/admin/**").hasRole("ADMIN")
                // 其余接口需要登录
                .anyRequest().authenticated()
            )

            // 异常处理
            .exceptionHandling(ex -> ex
                .authenticationEntryPoint((request, response, authException) -> {
                    response.setContentType("application/json;charset=utf-8");
                    response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
                    response.getWriter().write(
                        JSON.toJSONString(Result.error(401, "未登录或登录已过期"))
                    );
                })
                .accessDeniedHandler((request, response, accessDeniedException) -> {
                    response.setContentType("application/json;charset=utf-8");
                    response.setStatus(HttpServletResponse.SC_FORBIDDEN);
                    response.getWriter().write(
                        JSON.toJSONString(Result.error(403, "权限不足"))
                    );
                })
            )

            // 添加 JWT 过滤器
            .addFilterBefore(jwtAuthenticationFilter(),
                    UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }

    // 密码编码器
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();  // 使用 bcrypt 算法，不可逆加密
    }

    // JWT 过滤器
    @Bean
    public JwtAuthenticationFilter jwtAuthenticationFilter() {
        return new JwtAuthenticationFilter();
    }
}

1.3 认证（Authentication）

认证的核心是验证"你是谁"。

UserDetailsService 实现

@Service
public class UserDetailsServiceImpl implements UserDetailsService {

    @Autowired
    private UserMapper userMapper;

    @Override
    public UserDetails loadUserByUsername(String username)
            throws UsernameNotFoundException {

        // 1. 从数据库查询用户
        User user = userMapper.findByUsername(username);
        if (user == null) {
            throw new UsernameNotFoundException("用户不存在: " + username);
        }

        // 2. 查询用户角色和权限
        List<String> roles = userMapper.findRolesByUserId(user.getId());
        List<String> permissions = userMapper.findPermissionsByUserId(user.getId());

        // 3. 构建 UserDetails 对象
        return new org.springframework.security.core.userdetails.User(
                user.getUsername(),
                user.getPassword(),
                user.getEnabled(),
                true, true, true,  // 账号未过期、凭证未过期、账号未锁定
                buildAuthorities(roles, permissions)
        );
    }

    private Collection<? extends GrantedAuthority> buildAuthorities(
            List<String> roles, List<String> permissions) {
        Set<GrantedAuthority> authorities = new HashSet<>();

        // 角色（前缀 ROLE_）
        for (String role : roles) {
            authorities.add(new SimpleGrantedAuthority("ROLE_" + role));
        }

        // 权限
        for (String permission : permissions) {
            authorities.add(new SimpleGrantedAuthority(permission));
        }

        return authorities;
    }
}

自定义认证过滤器（JWT）

@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    @Autowired
    private JwtTokenProvider jwtTokenProvider;

    @Autowired
    private UserDetailsService userDetailsService;

    @Override
    protected void doFilterInternal(@NonNull HttpServletRequest request,
                                    @NonNull HttpServletResponse response,
                                    @NonNull FilterChain filterChain)
            throws ServletException, IOException {

        // 1. 从请求头获取 Token
        String token = extractToken(request);

        if (token != null && jwtTokenProvider.validateToken(token)) {
            // 2. 解析 Token 获取用户名
            String username = jwtTokenProvider.getUsernameFromToken(token);

            // 3. 加载用户信息
            UserDetails userDetails = userDetailsService.loadUserByUsername(username);

            // 4. 创建认证对象
            UsernamePasswordAuthenticationToken authentication =
                    new UsernamePasswordAuthenticationToken(
                            userDetails, null, userDetails.getAuthorities());
            authentication.setDetails(
                    new WebAuthenticationDetailsSource().buildDetails(request));

            // 5. 设置到 SecurityContext（标记已认证）
            SecurityContextHolder.getContext().setAuthentication(authentication);
        }

        filterChain.doFilter(request, response);
    }

    private String extractToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (bearerToken != null && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
}

基于方法注解的授权

@RestController
@RequestMapping("/api/users")
public class UserController {

    // 需要 user:read 权限
    @PreAuthorize("hasAuthority('user:read')")
    @GetMapping
    public Result<List<User>> list() {
        return Result.success(userService.list());
    }

    // 需要 user:create 权限
    @PreAuthorize("hasAuthority('user:create')")
    @PostMapping
    public Result<User> create(@Valid @RequestBody UserCreateRequest request) {
        return Result.success(userService.create(request));
    }

    // 只能操作自己的数据
    @PreAuthorize("hasAuthority('user:update') and #id == authentication.principal.id")
    @PutMapping("/{id}")
    public Result<User> update(@PathVariable Long id, @RequestBody UserUpdateRequest request) {
        return Result.success(userService.update(id, request));
    }

    // 多角色控制
    @PreAuthorize("hasRole('ADMIN') or hasRole('MANAGER')")
    @DeleteMapping("/{id}")
    public Result<Void> delete(@PathVariable Long id) {
        userService.delete(id);
        return Result.success();
    }

    // 复杂 SpEL 表达式
    @PreAuthorize("isAuthenticated() and (hasRole('ADMIN') or #userId == principal.id)")
    @GetMapping("/{userId}/profile")
    public Result<UserProfile> getProfile(@PathVariable Long userId) {
        return Result.success(userService.getProfile(userId));
    }
}

自定义权限校验

@Component("perm")
public class PermissionEvaluator {

    public boolean check(Long resourceUserId) {
        // 获取当前用户
        Long currentUserId = SecurityUtils.getCurrentUserId();
        // 只有资源所有者或管理员可操作
        return currentUserId.equals(resourceUserId) || hasAdminRole();
    }

    private boolean hasAdminRole() {
        return SecurityContextHolder.getContext()
                .getAuthentication().getAuthorities()
                .stream().anyMatch(a -> a.getAuthority().equals("ROLE_ADMIN"));
    }
}

// 使用
@PreAuthorize("@perm.check(#userId)")
@PostMapping("/{userId}/orders")

    return Result.success(orderService.create(userId, request));
}

二、JWT（JSON Web Token）

2.1 JWT 结构

JWT 由三部分组成，用 . 分隔：

JWT = Header.Payload.Signature

示例：
eyJhbGciOiJIUzI1NiJ9.eyJzdWIiOiJhZG1pbiIsInJvbGVzIjpbIkFETUlOIl19.4xUDFdW87HIiGUYvFlnAznf0z1wJ4qQHCyOq0QwJMAU

Header（头部）

{
  "alg": "HS256",       // 签名算法（HS256、RS256 等）
  "typ": "JWT"          // Token 类型
}

Payload（载荷）—— 不要放敏感信息！

{
  "sub": "admin",              // subject：用户标识（通常为用户 ID）
  "iat": 1705300000,           // issued at：签发时间
  "exp": 1705303600,           // expiration：过期时间
  "roles": ["ADMIN"],          // 自定义声明：角色
  "username": "admin"          // 自定义声明：用户名
}

Signature（签名）

HMACSHA256(
  base64UrlEncode(header) + "." + base64UrlEncode(payload),
  secret    // 服务端密钥，不可泄露
)

2.2 JWT 实现

@Component
public class JwtTokenProvider {

    // 密钥（实际应从配置中读取，建议使用 RSA 非对称加密）
    @Value("${jwt.secret}")
    private String jwtSecret;

    // Token 过期时间
    @Value("${jwt.expiration}")
    private long jwtExpiration;

    /**
     * 生成 Token
     */
    public String generateToken(UserDetails userDetails) {
        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + jwtExpiration);

        return Jwts.builder()
                .subject(userDetails.getUsername())                      // 用户名
                .claim("roles", userDetails.getAuthorities().stream()    // 角色
                        .map(GrantedAuthority::getAuthority)
                        .collect(Collectors.toList()))
                .issuedAt(now)                                           // 签发时间
                .expiration(expiryDate)                                  // 过期时间
                .signWith(getSigningKey())                               // 签名
                .compact();
    }

    /**
     * 从 Token 中提取用户名
     */
    public String getUsernameFromToken(String token) {
        return extractClaims(token).getSubject();
    }

    /**
     * 验证 Token 是否有效
     */
    public boolean validateToken(String token) {
        try {
            extractClaims(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            log.error("JWT Token 验证失败: {}", e.getMessage());
            return false;
        }
    }

    /**
     * 解析 Token 中的声明
     */
    private Claims extractClaims(String token) {
        return Jwts.parser()
                .verifyWith(getSigningKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    private SecretKey getSigningKey() {
        byte[] keyBytes = Decoders.BASE64.decode(jwtSecret);
        return Keys.hmacShaKeyFor(keyBytes);
    }
}

2.3 JWT 刷新机制

@Service
public class AuthService {

    @Autowired
    private JwtTokenProvider jwtTokenProvider;

    @Autowired
    private RefreshTokenRepository refreshTokenRepository;

    /**
     * 登录：返回 AccessToken + RefreshToken
     */
    public LoginResponse login(LoginRequest request) {
        // 1. 验证用户名密码
        Authentication authentication = authenticationManager.authenticate(
                new UsernamePasswordAuthenticationToken(
                        request.getUsername(), request.getPassword()));

        // 2. 生成 AccessToken（短期有效）
        String accessToken = jwtTokenProvider.generateToken(
                (UserDetails) authentication.getPrincipal());

        // 3. 生成 RefreshToken（长期有效，存数据库）
        String refreshToken = UUID.randomUUID().toString();
        refreshTokenRepository.save(new RefreshToken(
                refreshToken, request.getUsername(),
                LocalDateTime.now().plusDays(7)));  // 7 天有效期

        return new LoginResponse(accessToken, refreshToken);
    }

    /**
     * 刷新 AccessToken
     */
    public LoginResponse refresh(String refreshToken) {
        // 1. 验证 RefreshToken 有效性
        RefreshToken stored = refreshTokenRepository.findById(refreshToken)
                .orElseThrow(() -> new RuntimeException("无效的 RefreshToken"));

        if (stored.getExpiryDate().isBefore(LocalDateTime.now())) {
            throw new RuntimeException("RefreshToken 已过期，请重新登录");
        }

        // 2. 删除旧 Token（一次性使用）
        refreshTokenRepository.delete(stored);

        // 3. 生成新的 Token 对
        UserDetails userDetails = userDetailsService
                .loadUserByUsername(stored.getUsername());

        String newAccessToken = jwtTokenProvider.generateToken(userDetails);
        String newRefreshToken = UUID.randomUUID().toString();
        refreshTokenRepository.save(new RefreshToken(
                newRefreshToken, stored.getUsername(),
                LocalDateTime.now().plusDays(7)));

        return new LoginResponse(newAccessToken, newRefreshToken);
    }
}

2.4 JWT 优缺点

优点缺点
无状态（无需服务端存储 Session）Token 不可撤销（过期前始终有效）
跨域/跨服务共享Payload 不能存放敏感信息（Base64 非加密）
适合移动端和前后端分离Token 体积大（增加带宽开销）
客户端携带方便密钥管理复杂（泄露可伪造任意 Token）
解析速度快（不需查库）刷新机制增加复杂度

安全建议

  • 使用 HTTPS 传输 Token
  • Token 有效期不要太长（建议 15-30 分钟）
  • RefreshToken 妥善存储（建议使用 HttpOnly Cookie）
  • 实现 Token 黑名单机制（Redis 维护作废的 Token）
  • 敏感操作重新验证密码

三、OAuth2 基本流程

3.1 OAuth2 角色

+--------+                               +---------------+
|        | --> 请求授权 -->                |               |
| 用户   |                                | 资源所有者     |
| (User) | <-- 授权码/凭证 <--             | (Resource     |
|        |                                |  Owner)       |
+--------+                                +---------------+
    ↓                                             ↓
+-----------+                               +---------------+
| 客户端应用 | --> 获取 Token -->            |   授权服务器   |
| (Client)  | <-- 返回 Token  <--           | (Auth Server) |
+-----------+                               +---------------+
    ↓
+---------------+
|  资源服务器   |
| (Resource     |
|  Server)      |
+---------------+
角色说明示例
Resource Owner资源所有者（用户）你本人
Client第三方应用"使用微信登录"的 App
Authorization Server授权服务器微信 OAuth 服务
Resource Server资源服务器微信用户信息 API

3.2 授权码模式（最常用）

这是 OAuth2 最完整的流程，适用于有后端的 Web 应用。

用户访问第三方网站
    ↓
1. 客户端 → 授权服务器（请求授权码）
   GET /oauth2/authorize?response_type=code&client_id=APP_ID
       &redirect_uri=http://client/callback&scope=user_info
    ↓    用户登录并授权
2. 授权服务器 → 客户端（返回授权码）
   302 重定向到 http://client/callback?code=AUTH_CODE
    ↓
3. 客户端 → 授权服务器（用授权码换 Token）
   POST /oauth2/token
   grant_type=authorization_code&code=AUTH_CODE
   &redirect_uri=http://client/callback
   &client_id=APP_ID&client_secret=APP_SECRET
    ↓
4. 授权服务器 → 客户端（返回 AccessToken）
   { "access_token": "xxx", "token_type": "Bearer", "expires_in": 3600 }
    ↓
5. 客户端 → 资源服务器（用 Token 获取用户信息）
   GET /api/user
   Authorization: Bearer xxx
    ↓
6. 资源服务器 → 客户端（返回用户信息）
   { "id": 123, "name": "张三" }

3.3 密码模式

用户直接提供用户名密码给客户端，客户端用凭证换取 Token。

客户端 → 用户："请输入用户名和密码"
用户 → 客户端：用户名 + 密码
客户端 → 授权服务器：POST /oauth2/token
  grant_type=password&username=xxx&password=xxx&client_id=xxx
授权服务器 → 客户端：返回 AccessToken

适用场景：自家应用的登录（如微信 App 自身的登录） 不适用于：第三方应用接入（因为要共享密码）

3.4 客户端凭证模式

不需要用户参与，客户端用自己的凭证直接获取 Token。

客户端 → 授权服务器：POST /oauth2/token
  grant_type=client_credentials&client_id=xxx&client_secret=xxx
授权服务器 → 客户端：返回 AccessToken

适用场景：服务器间通信（内部微服务调用）

3.5 Spring Security OAuth2 客户端示例

spring:
  security:
    oauth2:
      client:
        registration:
          github:
            client-id: your-github-client-id
            client-secret: your-github-client-secret
            scope: user:email,read:user
          google:
            client-id: your-google-client-id
            client-secret: your-google-client-secret
            scope: profile,email
@Configuration
@EnableWebSecurity
public class OAuth2LoginConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
            .oauth2Login(oauth2 -> oauth2
                .loginPage("/login")                    // 自定义登录页
                .defaultSuccessUrl("/home", true)       // 登录成功后跳转
                .failureUrl("/login?error=true")        // 登录失败
                .userInfoEndpoint(userInfo -> userInfo
                    .userService(oAuth2UserService())   // 自定义用户处理
                )
            )
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/login", "/oauth2/**").permitAll()
                .anyRequest().authenticated()
            );
        return http.build();
    }

    @Bean
    public OAuth2UserService<OAuth2UserRequest, OAuth2User> oAuth2UserService() {
        return userRequest -> {
            // 默认实现
            DefaultOAuth2UserService delegate = new DefaultOAuth2UserService();
            OAuth2User oAuth2User = delegate.loadUser(userRequest);

            String registrationId = userRequest.getClientRegistration().getRegistrationId();
            String userNameAttributeName = userRequest.getClientRegistration()
                    .getProviderDetails().getUserInfoEndpoint()
                    .getUserNameAttributeName();

            // 从 OAuth2 用户信息中提取数据，创建/查询本地用户
            Map<String, Object> attributes = oAuth2User.getAttributes();
            String email = (String) attributes.get("email");
            String name = (String) attributes.get("name");

            // 查找或创建本地用户
            User user = userService.findOrCreateByOAuth2(registrationId, email, name);

            // 返回包装后的 OAuth2User
            return new CustomOAuth2User(user, attributes, userNameAttributeName);
        };
    }
}

项目 1：Spring Security + JWT 登录认证

目标：实现完整的用户认证和授权系统。

功能要求：

  1. 用户注册（密码使用 BCrypt 加密）
  2. 用户登录（返回 JWT Token）
  3. Token 刷新机制（AccessToken + RefreshToken）
  4. 基于 URL 和方法的权限控制
  5. 自定义未登录和权限不足的响应

项目 2：OAuth2 第三方登录

目标：集成 GitHub/Google OAuth2 登录。

功能要求：

  1. 支持 GitHub 和 Google 账号登录
  2. 首次登录自动注册本地账号
  3. 绑定本地账号和第三方账号
  4. 登录后颁发 JWT Token
