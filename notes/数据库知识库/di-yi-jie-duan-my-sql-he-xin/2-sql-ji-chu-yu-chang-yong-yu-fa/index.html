2-SQL基础与常用语法

约 1480 字大约 5 分钟

2026-08-18

一、SQL 分类

分类全称作用常见语句
DDLData Definition Language定义数据库对象（库、表、索引）CREATE, ALTER, DROP, TRUNCATE
DMLData Manipulation Language数据增删改INSERT, UPDATE, DELETE
DQLData Query Language数据查询SELECT, WHERE, GROUP BY, HAVING, ORDER BY
DCLData Control Language权限与事务控制GRANT, REVOKE, COMMIT, ROLLBACK

二、DDL — 数据定义

2.1 数据库操作

-- 创建数据库（指定字符集）
CREATE DATABASE IF NOT EXISTS mydb
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

-- 查看所有数据库
SHOW DATABASES;

-- 切换数据库
USE mydb;

-- 删除数据库
DROP DATABASE IF EXISTS mydb;

2.2 表操作

-- 创建表
CREATE TABLE IF NOT EXISTS users (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  username    VARCHAR(50)     NOT NULL UNIQUE,
  email       VARCHAR(100)    NOT NULL,
  password    VARCHAR(255)    NOT NULL,
  status      TINYINT         NOT NULL DEFAULT 1 COMMENT '1=启用 0=禁用',
  created_at  DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at  DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  INDEX idx_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 查看表结构
DESC users;
SHOW CREATE TABLE users;

-- 修改表
ALTER TABLE users ADD COLUMN phone VARCHAR(20) AFTER email;     -- 添加列
ALTER TABLE users MODIFY COLUMN phone VARCHAR(30);              -- 修改列定义
ALTER TABLE users DROP COLUMN phone;                            -- 删除列
ALTER TABLE users ADD INDEX idx_status (status);                -- 添加索引
ALTER TABLE users RENAME TO members;                            -- 重命名表

-- 删除表
DROP TABLE IF EXISTS users;
TRUNCATE TABLE users;  -- 清空表数据（不保留自增计数）

三、DML — 数据增删改

3.1 INSERT

-- 单行插入
INSERT INTO users (username, email, password)
VALUES ('zhangsan', 'zhangsan@example.com', 'hashed_pwd');

-- 多行插入（推荐，性能更好）
INSERT INTO users (username, email, password) VALUES
  ('lisi',    'lisi@example.com',    'hashed_pwd'),
  ('wangwu',  'wangwu@example.com',  'hashed_pwd');

-- INSERT ... ON DUPLICATE KEY UPDATE（存在则更新，不存在则插入）
INSERT INTO users (username, email, password)
VALUES ('zhangsan', 'new_email@example.com', 'new_pwd')
ON DUPLICATE KEY UPDATE email = VALUES(email), password = VALUES(password);

-- REPLACE INTO（存在则删除再插入，会触发 DELETE + INSERT）
REPLACE INTO users (id, username, email, password)
VALUES (1, 'zhangsan', 'fixed@example.com', 'pwd');

3.2 UPDATE

-- 基本更新
UPDATE users SET status = 0 WHERE id = 1;

-- 多列更新
UPDATE users
SET status = 0, updated_at = NOW()
WHERE id = 1;

-- 关联更新（用 JOIN）
UPDATE users u
JOIN orders o ON u.id = o.user_id
SET u.status = 0
WHERE o.created_at < '2025-01-01';

-- 限制更新行数（防止误更新）
UPDATE users SET status = 0 WHERE created_at < '2025-01-01' LIMIT 100;

3.3 DELETE

-- 基本删除
DELETE FROM users WHERE id = 1;

-- 关联删除
DELETE u FROM users u
JOIN orders o ON u.id = o.user_id
WHERE o.status = 'cancelled';

-- 限制删除行数
DELETE FROM users WHERE status = 0 LIMIT 50;

-- 注意：DELETE 不会重置 AUTO_INCREMENT，TRUNCATE 会

安全提醒：执行 UPDATE/DELETE 前，先用相同的 WHERE 条件跑一次 SELECT 确认影响范围，避免误操作。

四、DQL — 数据查询

4.1 基础查询

-- 选择指定列
SELECT id, username, email FROM users;

-- 去重
SELECT DISTINCT status FROM users;

-- 别名
SELECT id AS user_id, username AS name FROM users;

-- 条件过滤
SELECT * FROM users WHERE status = 1 AND created_at >= '2025-01-01';

-- IN / NOT IN
SELECT * FROM users WHERE id IN (1, 2, 3);

-- BETWEEN
SELECT * FROM users WHERE created_at BETWEEN '2025-01-01' AND '2025-12-31';

-- LIKE（模糊查询）
SELECT * FROM users WHERE username LIKE 'zhang%';    -- 以 zhang 开头
SELECT * FROM users WHERE email LIKE '%example.com'; -- 以 example.com 结尾

-- IS NULL / IS NOT NULL
SELECT * FROM users WHERE phone IS NULL;

4.2 ORDER BY 与 LIMIT

-- 排序（ASC 升序，DESC 降序）
SELECT * FROM users ORDER BY created_at DESC;

-- 多字段排序
SELECT * FROM users ORDER BY status ASC, created_at DESC;

-- 分页（offset 从 0 开始）
SELECT * FROM users ORDER BY id LIMIT 10 OFFSET 20;  -- 第 21-30 条
SELECT * FROM users ORDER BY id LIMIT 20, 10;          -- 同上（MySQL 方言）

4.3 JOIN 连接查询

-- INNER JOIN（内连接，只返回匹配行）
SELECT u.username, o.order_no, o.total
FROM users u
INNER JOIN orders o ON u.id = o.user_id;

-- LEFT JOIN（左连接，保留左表全部行）
SELECT u.username, COUNT(o.id) AS order_count
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id;

-- RIGHT JOIN（右连接，保留右表全部行）
SELECT u.username, o.order_no
FROM users u
RIGHT JOIN orders o ON u.id = o.user_id;

-- CROSS JOIN（笛卡尔积）
SELECT u.username, p.product_name
FROM users u
CROSS JOIN products p;

-- 自连接（同一表连接自身）
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

4.4 GROUP BY 与 HAVING

-- 分组统计
SELECT user_id, COUNT(*) AS order_count, SUM(total) AS total_amount
FROM orders
GROUP BY user_id;

-- HAVING 过滤分组结果（WHERE 过滤行，HAVING 过滤组）
SELECT user_id, SUM(total) AS total_amount
FROM orders
WHERE status = 'paid'
GROUP BY user_id
HAVING total_amount > 1000;

-- 常见错误：SELECT 中包含非聚合列且不在 GROUP BY 中
-- MySQL 默认允许（非标准），严格模式下会报错
-- 正确做法：所有非聚合列都应出现在 GROUP BY 中

4.5 子查询

-- 标量子查询（返回单值）
SELECT * FROM users WHERE id = (SELECT user_id FROM orders ORDER BY total DESC LIMIT 1);

-- IN 子查询
SELECT * FROM users WHERE id IN (SELECT DISTINCT user_id FROM orders WHERE total > 500);

-- EXISTS 子查询（推荐：大表关联时性能优于 IN）
SELECT * FROM users u WHERE EXISTS (
  SELECT 1 FROM orders o WHERE o.user_id = u.id AND o.status = 'paid'
);

-- 派生表（FROM 子查询）
SELECT t.user_id, t.order_count
FROM (
  SELECT user_id, COUNT(*) AS order_count
  FROM orders
  GROUP BY user_id
) t
WHERE t.order_count > 5;

五、常用内置函数

5.1 字符串函数

SELECT CONCAT('Hello', ' ', 'World');        -- 'Hello World'
SELECT CONCAT_WS(',', 'a', 'b', 'c');        -- 'a,b,c'
SELECT LENGTH('hello');                        -- 5（字节数）
SELECT CHAR_LENGTH('你好');                     -- 2（字符数）
SELECT UPPER('hello');                         -- 'HELLO'
SELECT LOWER('HELLO');                         -- 'hello'
SELECT SUBSTRING('Hello World', 1, 5);        -- 'Hello'
SELECT REPLACE('Hello World', 'World', 'MySQL'); -- 'Hello MySQL'
SELECT TRIM('  hello  ');                      -- 'hello'
SELECT LPAD('42', 5, '0');                    -- '00042'
SELECT RPAD('hi', 5, '.');                    -- 'hi...'

5.2 日期函数

SELECT NOW();                                  -- 当前日期时间
SELECT CURDATE();                              -- 当前日期
SELECT CURTIME();                              -- 当前时间
SELECT DATE_FORMAT(NOW(), '%Y-%m-%d %H:%i:%s'); -- 格式化
SELECT DATEDIFF('2025-12-31', '2025-01-01');   -- 364（天数差）
SELECT DATE_ADD(NOW(), INTERVAL 7 DAY);        -- 加 7 天
SELECT DATE_SUB(NOW(), INTERVAL 1 MONTH);      -- 减 1 个月
SELECT YEAR(NOW());                            -- 2026
SELECT MONTH(NOW());                           -- 8

5.3 聚合函数

SELECT
  COUNT(*)          AS total,
  COUNT(DISTINCT user_id) AS unique_users,
  SUM(total)        AS total_amount,
  AVG(total)        AS avg_amount,
  MAX(total)        AS max_amount,
  MIN(total)        AS min_amount
FROM orders;

六、EXPLAIN — 查询执行计划

EXPLAIN 是分析慢查询最重要的工具，下一章会深入讲解，这里先介绍基本用法：

EXPLAIN SELECT * FROM users WHERE email = 'test@example.com';
字段含义关注点
type访问类型ALL（全表扫描）最差，const/eq_ref 最优
key实际使用的索引NULL 表示没有使用索引
rows预估扫描行数越小越好
