5-获取当前上下文信息-内置函数与常量

约 1513 字大约 5 分钟

2026-08-18

一、MySQL 上下文信息函数

1.1 当前用户与会话

-- 当前登录用户名
SELECT CURRENT_USER();      -- 'app_user'@'%'（含来源主机）
SELECT USER();              -- 同 CURRENT_USER()，MySQL 别名
SELECT SESSION_USER();      -- 同 CURRENT_USER()

-- 当前会话用户名（不含主机）
SELECT CURRENT_USER();

-- 连接 ID（每个连接唯一，用于 KILL 终止）
SELECT CONNECTION_ID();      -- 12345

-- 当前数据库
SELECT DATABASE();           -- 'mydb'（未选择则返回 NULL）
SELECT SCHEMA();             -- 同 DATABASE()

-- 服务器版本
SELECT VERSION();            -- '8.0.39-0ubuntu0.22.04.1'

1.2 服务器状态与配置

-- 查看服务器状态变量（运行时计数器）
SHOW STATUS LIKE 'Threads_connected';   -- 当前连接数
SHOW STATUS LIKE 'Questions';           -- 已执行查询总数
SHOW STATUS LIKE 'Uptime';             -- 运行时长（秒）
SHOW STATUS LIKE 'Slow_queries';       -- 慢查询总数
SHOW STATUS LIKE 'Innodb_rows_read';   -- InnoDB 行读取数

-- 查看系统变量（配置参数）
SHOW VARIABLES LIKE 'max_connections';  -- 最大连接数
SHOW VARIABLES LIKE 'innodb_buffer_pool_size';  -- Buffer Pool 大小
SHOW VARIABLES LIKE 'tx_isolation';    -- 当前事务隔离级别

-- 查看指定变量的当前值
SELECT @@GLOBAL.max_connections;        -- 全局值
SELECT @@SESSION.tx_isolation;          -- 会话值
SELECT @@GLOBAL.max_connections = @@SESSION.max_connections;  -- 是否一致

1.3 时间与日期

-- 当前时间
SELECT NOW();                -- 2026-08-18 14:30:00
SELECT CURRENT_TIMESTAMP();  -- 同 NOW()
SELECT SYSDATE();            -- 实时时间（函数执行时刻，非语句开始时刻）
SELECT UTC_TIMESTAMP();      -- UTC 时间

-- 当前日期 / 时间
SELECT CURDATE();            -- 2026-08-18
SELECT CURTIME();            -- 14:30:00

-- Unix 时间戳
SELECT UNIX_TIMESTAMP();             -- 当前时间戳（秒）
SELECT UNIX_TIMESTAMP('2026-08-18'); -- 指定日期的时间戳
SELECT FROM_UNIXTIME(1723977000);    -- 时间戳转日期

1.4 事务信息

-- 当前事务是否活跃
SELECT IF(@@in_transaction = 1, '活跃', '无事务');  -- 0=无 1=有

-- 最后一次自增 ID（当前会话）
SELECT LAST_INSERT_ID();    -- 返回当前会话最近一次 INSERT 的自增值

-- 受影响的行数（上一条 DML）
SELECT ROW_COUNT();         -- 上一条 UPDATE/DELETE/INSERT 影响的行数

1.5 权限相关

-- 查看当前用户权限
SHOW GRANTS;
SHOW GRANTS FOR 'app_user'@'%';  -- 查看指定用户

-- 检查是否有某权限
SELECT HAS_DATABASE_PRIVILEGE('app_user', 'mydb', 'SELECT');  -- 1/0
SELECT HAS_TABLE_PRIVILEGE('app_user', 'mydb.users', 'INSERT');

二、PostgreSQL 上下文信息函数

2.1 当前用户与会话

-- 当前用户名
SELECT current_user;            -- 'app_user'（角色名）
SELECT session_user;            -- 会话用户（通常与 current_user 相同）

-- 当前数据库
SELECT current_database();      -- 'mydb'

-- 当前 schema
SELECT current_schema();        -- 'public'
SELECT current_schemas(true);   -- 搜索路径中所有 schema：{public, "$user"}

-- 服务器版本
SELECT version();
-- PostgreSQL 16.4 on x86_64-pc-linux-gnu, compiled by gcc...

-- 连接 PID（用于 pg_cancel_backend / pg_terminate_backend）
SELECT pg_backend_pid();        -- 12345

-- 当前连接的客户端地址和端口
SELECT inet_client_addr();      -- 127.0.0.1（远程IP）
SELECT inet_client_port();      -- 54321
SELECT inet_server_addr();      -- 服务端监听 IP
SELECT inet_server_port();      -- 5432

1.2 服务器配置与状态

-- 查看当前会话配置值
SHOW work_mem;                 -- 16MB
SHOW shared_buffers;
SHOW max_connections;

-- 查看全局配置
SELECT setting, unit, short_desc
FROM pg_settings
WHERE name = 'shared_buffers';

-- 查看所有配置（当前连接生效值 vs 默认值）
SELECT name, setting, unit, boot_val, reset_val
FROM pg_settings
WHERE name LIKE '%work_mem%';

-- 查看服务器运行状态
SELECT
  datname,
  numbackends AS connections,
  xact_commit AS commits,
  xact_rollback AS rollbacks,
  blks_read,
  blks_hit,
  ROUND(blks_hit * 100.0 / NULLIF(blks_hit + blks_read, 0), 2) AS cache_hit_pct
FROM pg_stat_database
WHERE datname = current_database();

-- 查看当前连接统计
SELECT state, COUNT(*)
FROM pg_stat_activity
GROUP BY state;

1.3 时间与日期

-- 当前时间
SELECT NOW();                      -- 2026-08-18 14:30:00+08
SELECT CURRENT_TIMESTAMP;          -- 同 NOW()
SELECT CURRENT_TIMESTAMP AT TIME ZONE 'UTC';  -- 转为 UTC
SELECT LOCALTIMESTAMP;             -- 无时区信息
SELECT clock_timestamp();          -- 实时时间（函数执行时刻）

-- 当前日期 / 时间
SELECT CURRENT_DATE;               -- 2026-08-18
SELECT CURRENT_TIME;               -- 14:30:00+08
SELECT LOCALTIME;                  -- 14:30:00（无时区）

-- Unix 时间戳
SELECT EXTRACT(EPOCH FROM NOW());                    -- 当前时间戳（秒，numeric）
SELECT TO_TIMESTAMP(1723977000);                     -- 时间戳转日期
SELECT EXTRACT(EPOCH FROM NOW())::bigint;            -- 转为 bigint

-- 日期截断与提取
SELECT date_trunc('hour', NOW());          -- 截断到小时
SELECT EXTRACT(dow FROM NOW());            -- 星期几（0=周日）
SELECT EXTRACT(isoyear FROM NOW());        -- ISO 年

1.4 事务信息

-- 当前事务 ID
SELECT txid_current();                    -- 当前事务 ID（未开启则自动开启事务）
SELECT txid_current_snapshot();           -- 当前快照：'100:105:100,102,103'
-- 格式：xmin:xmax:xip_list
-- xmin: 该快照中所有 xmin < xmin 的事务已提交
-- xmax: 该快照中所有 xmax > xmax 的事务未开始
-- xip_list: 该快照中正在活跃的事务 ID 列表

-- 当前事务是否只读
SELECT current_setting('transaction_read_only');  -- 'on' / 'off'

1.5 权限相关

-- 当前用户是否具有某角色
SELECT current_setting('role');  -- 当前角色

-- 查看用户权限
SELECT
  grantee,
  privilege_type,
  table_name
FROM information_schema.role_table_grants
WHERE grantee = current_user;

-- 检查当前用户是否是超级用户
SELECT current_setting('is_superuser');  -- 'on' / 'off'

-- 查看表权限
\dp users  -- psql 元命令

1.6 数据库对象信息

-- 查看当前 schema 下所有表
SELECT tablename FROM pg_tables WHERE schemaname = current_schema();

-- 查看表的列信息
SELECT
  column_name,
  data_type,
  is_nullable,
  column_default
FROM information_schema.columns
WHERE table_name = 'users' AND table_schema = current_schema();

-- 查看索引
SELECT
  indexname,
  indexdef
FROM pg_indexes
WHERE tablename = 'users' AND schemaname = current_schema();

-- 查看视图
SELECT viewname FROM pg_views WHERE schemaname = current_schema();

-- 查看函数
SELECT
  p.proname AS function_name,
  pg_catalog.pg_get_function_result(p.oid) AS return_type,
  pg_catalog.pg_get_function_arguments(p.oid) AS arguments
FROM pg_proc p
JOIN pg_namespace n ON p.pronamespace = n.oid
WHERE n.nspname = current_schema();

四、实用查询速查

4.1 快速了解当前连接环境

MySQL：

SELECT
  CONNECTION_ID()       AS conn_id,
  USER()                AS user,
  DATABASE()            AS `database`,
  VERSION()             AS mysql_version,
  @@tx_isolation        AS isolation,
  @@character_set_client AS charset,
  NOW()                 AS current_time;

PostgreSQL：

SELECT
  pg_backend_pid()      AS conn_pid,
  current_user          AS username,
  current_database()    AS database_name,
  version()             AS pg_version,
  current_schema()      AS schema_name,
  current_setting('transaction_isolation') AS isolation,
  inet_client_addr()    AS client_ip,
  NOW()                 AS current_time;

4.2 监控慢会话

MySQL：

SELECT id, user, host, db, command, time, state, info
FROM information_schema.PROCESSLIST
WHERE command != 'Sleep' AND time > 60
ORDER BY time DESC;

PostgreSQL：

SELECT
  pid,
  usename,
  datname,
  client_addr,
  state,
  AGE(NOW(), query_start) AS duration,
  LEFT(query, 80) AS query
FROM pg_stat_activity
WHERE state = 'active'
  AND query_start < NOW() - INTERVAL '60 seconds'
ORDER BY query_start;
